    65→interface MealsCacheValue {
    66→  data: MealInstanceDto[];
    67→  meta: MealsMeta;
    68→}
    69→
    70→function patchCacheRows(
    71→  value: MealsCacheValue | undefined,
    72→  instanceId: string,
    73→  patch: (m: MealInstanceDto) => MealInstanceDto
    74→): MealsCacheValue | undefined {
    75→  if (!value) return value;
    76→  return { ...value, data: value.data.map((m) => (m.id === instanceId ? patch(m) : m)) };
    77→}

    79→/* ------------------------------- date helpers ------------------------------ */

    81→/** (year, monthIndex 0-11) → "YYYY-MM". */
    82→function monthKeyOf(year: number, month: number): string {
    83→  return `${year}-${String(month + 1).padStart(2, "0")}`;
    84→}
    85→
    86→function monthStartKey(year: number, month: number): string {
    87→  return `${monthKeyOf(year, month)}-01`;
    88→}
    89→
    90→function monthEndKey(year: number, month: number): string {
    91→  const lastDay = new Date(Date.UTC(year, month + 1, 0)).getUTCDate();
    92→  return `${monthKeyOf(year, month)}-${String(lastDay).padStart(2, "0")}`;
    93→}
    94→
    95→/** "September 2025". */
    96→function monthLongLabel(year: number, month: number): string {
    97→  return new Intl.DateTimeFormat("en-IN", { month: "long", year: "numeric", timeZone: "UTC" }).format(
    98→    new Date(Date.UTC(year, month, 1))
    99→  );
   100→}
   101→
   102→/** "Sep 2025" — compact KPI sub for non-current months. */
   103→function monthShortKey(year: number, month: number): string {
   104→  return new Intl.DateTimeFormat("en-IN", { month: "short", year: "numeric", timeZone: "UTC" }).format(
   105→    new Date(Date.UTC(year, month, 1))
   106→  );
   107→}
   108→
   109→/** Picker pill labels — relative for ±1 day, compact otherwise (BoardOps pattern). */
   110→function dayPillLabels(key: string, todayKey: string): { top: string; bottom: string } {
   111→  if (key === todayKey) return { top: "Today", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   112→  if (key === addDaysToKey(todayKey, -1)) return { top: "Yesterday", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   113→  if (key === addDaysToKey(todayKey, 1)) return { top: "Tomorrow", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   114→  return { top: formatDateInTz(key, "UTC"), bottom: weekdayShort(key) };
   115→}

   216→  /* --------------------------- optimistic toggle --------------------------- */
   217→
   218→  async function handleToggle(meal: MealInstanceDto, next: "ON" | "OFF") {
   219→    const queries = queryClient.getQueriesData<MealsCacheValue>({
   220→      queryKey: ["api", "/api/v1/meals"],
   221→    });
   222→    const snapshots = queries.map(([key, value]) => ({ key, value }));
   223→
   224→    let current: MealInstanceDto | undefined;
   225→    for (const { value } of snapshots) {
   226→      current = value?.data.find((m) => m.id === meal.id);
   227→      if (current) break;
   228→    }
   229→    if (!current) return;
   230→
   231→    const expectedVersion = current.myState.version;
   232→
   233→    // Optimistically flip every cached copy of this row.
   234→    for (const { key } of snapshots) {
   235→      queryClient.setQueryData<MealsCacheValue>(key, (old) =>
   236→        patchCacheRows(old, meal.id, (m) => ({
   237→          ...m,
   238→          myState: { ...m.myState, effectiveState: next } as MealMyState,
   239→        }))
   240→      );
   241→    }
   242→
   243→    try {
   244→      const res = await apiJson<ToggleResponse>(`/api/v1/meals/${meal.id}/toggle`, "POST", {
   245→        state: next,
   246→        expectedVersion,
   247→      });
   248→      // Authoritative response wins.
   249→      for (const { key } of snapshots) {
   250→        queryClient.setQueryData<MealsCacheValue>(key, (old) =>
   251→          patchCacheRows(old, meal.id, (m) => ({
   252→            ...m,
   253→            myState: {
   254→              ...m.myState,
   255→              effectiveState: res.state,
   256→              effectiveReason: res.effectiveReason,
   257→              locked: res.locked,
   258→              version: res.version,
   259→            } as MealMyState,
   260→          }))
   261→        );
   262→      }
   263→      invalidate([RESIDENT_KEYS.dashboard]);
   264→    } catch (err) {
   265→      // Rollback — the toggle springs back (spec §114).
   266→      for (const { key, value } of snapshots) {
   267→        queryClient.setQueryData<MealsCacheValue>(key, value);
   268→      }
   269→      if (err instanceof ApiClientError) {
   270→        if (err.code === "MEAL_CUTOFF_PASSED") {
   271→          flashFor(meal.id, {
   272→            tone: "warning",
   273→            text: `${meal.name} is already locked. Admin can still override it.`,
   274→          });
   275→        } else if (err.code === "RESOURCE_CHANGED") {
   276→          flashFor(meal.id, { tone: "info", text: "This meal was just changed. Refreshing…" });
   277→          void queryClient.invalidateQueries({ queryKey: ["api", "/api/v1/meals"] });
   278→        } else if (err.code === "MEAL_NOT_AVAILABLE") {
   279→          flashFor(meal.id, { tone: "warning", text: "This meal isn't available to change right now." });
   280→        } else {
   281→          flashFor(meal.id, { tone: "danger", text: err.message });
   282→        }
   283→      } else {
   284→        flashFor(meal.id, { tone: "danger", text: friendlyError(err) });
   285→      }
   286→    }
   287→  }
