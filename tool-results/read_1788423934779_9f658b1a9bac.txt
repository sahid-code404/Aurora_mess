   108→function draftFromDef(def: MealDefinitionRow): DraftDef {
   109→  return {
   110→    name: def.name,
   111→    description: def.description ?? "",
   112→    icon: def.icon ?? "utensils",
   113→    colorToken: def.colorToken ?? "teal",
   114→    mealType: def.mealType,
   115→    scheduleStrategy: (def.scheduleStrategy as DraftDef["scheduleStrategy"]) ?? "DAILY",
   116→    weekdays: (def.weekdaysCsv ?? "").split(",").map((s) => Number(s.trim())).filter((n) => n >= 1 && n <= 7),
   117→    specificDate: def.specificDate ?? "",
   118→    serviceStartLocal: def.serviceStartLocal,
   119→    serviceEndLocal: def.serviceEndLocal,
   120→    cutoffStrategy: (def.cutoffStrategy as DraftDef["cutoffStrategy"]) ?? "SAME_DAY",
   121→    cutoffOffsetDays: String(def.cutoffOffsetDays ?? 0),
   122→    cutoffLocalTime: def.cutoffLocalTime,
   123→    pricingStrategy: (def.pricingStrategy as DraftDef["pricingStrategy"]) ?? "FORMULA",
   124→    fixedPriceMinor: def.fixedPriceMinor != null ? (def.fixedPriceMinor / 100).toFixed(2) : "",
   125→    defaultState: def.defaultState === "OFF" ? "OFF" : "ON",
   126→    defaultVisible: def.defaultVisible,
   127→    internalNotes: def.internalNotes ?? "",
   128→  };
   129→}

   131→const STEPS = ["Basics", "Schedule", "Cutoff", "Pricing", "Defaults", "Review"];
   132→const HHMM = /^([01]\d|2[0-3]):[0-5]\d$/;
   133→
   134→function stepProblem(draft: DraftDef, step: number): string | null {
   135→  switch (step) {
   136→    case 0:
   137→      if (draft.name.trim().length < 2) return "Give the meal a name (at least 2 characters).";
   138→      return null;
   139→    case 1: {
   140→      if (!HHMM.test(draft.serviceStartLocal) || !HHMM.test(draft.serviceEndLocal))
   141→        return "Set the service window as HH:MM.";
   142→      if (draft.serviceStartLocal >= draft.serviceEndLocal) return "Service end must be after service start.";
   143→      if (draft.scheduleStrategy === "WEEKDAYS" && draft.weekdays.length === 0) return "Pick at least one weekday.";
   144→      if (draft.scheduleStrategy === "ONE_TIME" && !draft.specificDate) return "Pick the one-time date.";
   145→      return null;
   146→    }
   147→    case 2:
   148→      if (!HHMM.test(draft.cutoffLocalTime)) return "Set the cutoff time as HH:MM.";
   149→      if (draft.cutoffStrategy === "CUSTOM_OFFSET" && !/^\d{1,2}$/.test(draft.cutoffOffsetDays.trim()))
   150→        return "Enter the offset in days (0–30).";
   151→      return null;
   152→    case 3:
   153→      if (draft.pricingStrategy === "FIXED" && moneyProblem(draft.fixedPriceMinor)) return "Enter a fixed price like 55.00.";
   154→      return null;
   155→    default:
   156→      return null;
   157→  }
   158→}
   159→
   160→function draftToPayload(draft: DraftDef): Record<string, unknown> {
   161→  const payload: Record<string, unknown> = {
   162→    name: draft.name.trim(),
   163→    description: draft.description.trim() || undefined,
   164→    icon: draft.icon,
   165→    colorToken: draft.colorToken,
   166→    mealType: draft.mealType,
   167→    defaultState: draft.defaultState,
   168→    defaultVisible: draft.defaultVisible,
   169→    pricingStrategy: draft.pricingStrategy,
   170→    scheduleStrategy: draft.scheduleStrategy,
   171→    serviceStartLocal: draft.serviceStartLocal,
   172→    serviceEndLocal: draft.serviceEndLocal,
   173→    cutoffStrategy: draft.cutoffStrategy,
   174→    cutoffLocalTime: draft.cutoffLocalTime,
   175→    internalNotes: draft.internalNotes.trim() || undefined,
   176→  };
   177→  if (draft.scheduleStrategy === "WEEKDAYS") {
   178→    payload.weekdaysCsv = [...draft.weekdays].sort((a, b) => a - b).join(",");
   179→  }
   180→  if (draft.scheduleStrategy === "ONE_TIME") payload.specificDate = draft.specificDate;
   181→  if (draft.cutoffStrategy === "CUSTOM_OFFSET") payload.cutoffOffsetDays = Number(draft.cutoffOffsetDays);
   182→  if (draft.pricingStrategy === "FIXED") payload.fixedPriceMinor = draft.fixedPriceMinor.trim();
   183→  return payload;
   184→}

   471→        </motion.div>

   501→      invalidate([DEFS_PATH]);
   502→      toast.success("Meal archived", {
   503→        description: `${target.name} stops generating future services. History stays intact.`,
   504→      });
   505→      setArchiveTarget(null);
   506→    } catch (err) {
   507→      toast.error(errMessage(err));
   508→    } finally {
   509→      setActing(false);
   510→    }
   511→  }
   512→
   513→  async function requestDeletion(target: MealDefinitionRow, reason: string | undefined) {
   514→    setActing(true);
   515→    try {
   516→      await postJson(`${DEFS_PATH}/${target.id}/request-deletion`, { reason });
   517→      invalidate([DEFS_PATH]);
   518→      toast.success("Deletion queued", {
   519→        description: `${target.name} will be removed after the 30-day safety window.`,
   520→      });
   521→      setDeleteTarget(null);
   522→    } catch (err) {
   523→      toast.error(errMessage(err));
   524→    } finally {
   525→      setActing(false);
   526→    }
   527→  }
   528→
   529→  if (isLoading) {
   530→    return (
   531→      <div className="space-y-6">
   532→        <PageHeader title="Meal Configuration" subtitle="Define services, cutoffs and defaults — every edit is versioned." />
   533→        <KpiGridSkeleton count={3} />
   534→        <ListSkeleton rows={3} />
   535→      </div>
   536→    );
   537→  }
   538→
   539→  if (error || !envelope) {
   540→    return (
   541→      <div className="space-y-6">
   542→        <PageHeader title="Meal Configuration" subtitle="Define services, cutoffs and defaults — every edit is versioned." />
   543→        <ErrorState
   544→          code={(error as ApiClientError | undefined)?.code}
   545→          message={(error as ApiClientError | undefined)?.message}
   546→          onRetry={() => void refetch()}
   547→        />
   548→      </div>
   549→    );
   550→  }
   551→
   552→  return (
   553→    <div className="space-y-5">
   554→      <PageHeader
   555→        title="Meal Configuration"
   556→        subtitle="Define services, cutoffs and defaults — every edit is versioned."
   557→        actions={
   558→          <GlassButton
   559→            variant="primary"
   560→            icon={<Plus />}
   561→            onClick={() => {
   562→              setEditingDef(null);
   563→              setWizardOpen(true);
   564→            }}
   565→          >
   566→            New Meal
   567→          </GlassButton>
   568→        }
   569→      />
   570→
   571→      <KpiGrid
   572→        kpis={[
   573→          { label: "Configured", value: String(metaNum(meta, "configured") ?? defs.length), icon: <Settings2 />, sub: "Total definitions" },
   574→          { label: "Active", value: String(metaNum(meta, "active") ?? "—"), icon: <Utensils />, sub: "Generating services" },
   575→          { label: "Inactive", value: String(metaNum(meta, "inactive") ?? "—"), icon: <Archive />, sub: "Archived" },
   576→        ]}
   577→      />
   578→
   579→      {defs.length === 0 ? (
   580→        <EmptyState
   581→          icon={Utensils}
   582→          title="No meals configured"
   583→          description="Create your first meal definition — residents see and opt into meals from here."
   584→          action={
   585→            <GlassButton variant="primary" icon={<Plus />} onClick={() => setWizardOpen(true)}>
   586→              New Meal
   587→            </GlassButton>
   588→          }
   589→        />
   590→      ) : (
   591→        <div className="grid grid-cols-1 gap-3 lg:grid-cols-2 xl:grid-cols-3">
   592→          <AnimatePresence initial={false}>
   593→            {defs.map((def) => {
   594→              const Icon = mealIcon(def.icon);
   595→              const inactive = def.archivedAt != null;
   596→              return (
   597→                <motion.div
   598→                  key={def.id}
   599→                  layout
   600→                  initial={{ opacity: 0, y: 8 }}
   601→                  animate={{ opacity: 1, y: 0 }}
   602→                  exit={{ opacity: 0, y: -4 }}
   603→                  transition={{ duration: 0.22, ease: "easeOut" }}
   604→                >
   605→                  <GlassCard className="p-4 sm:p-5">
   606→                    <div className="flex items-start justify-between gap-3">
   607→                      <div className="flex min-w-0 items-center gap-3">
   608→                        <span className="glass-inset flex size-10 shrink-0 items-center justify-center rounded-md text-primary [&_svg]:size-[18px]">
   609→                          <Icon aria-hidden />
   610→                        </span>
   611→                        <div className="min-w-0">
   612→                          <div className="flex flex-wrap items-center gap-1.5">
   613→                            <p className="truncate text-sm font-semibold">{def.name}</p>
   614→                            {!def.defaultVisible && <Chip tone="neutral">Hidden</Chip>}
   615→                          </div>
   616→                          <p className="truncate text-[11px] text-muted-foreground">
   617→                            {def.mealType.toLowerCase()} · v{def.latestVersion?.version ?? 1}
   618→                          </p>
   619→                        </div>
   620→                      </div>
   621→                      <StatusBadge status={inactive ? "INACTIVE" : "ACTIVE"} />
   622→                    </div>
   623→
   624→                    {def.description && <p className="mt-2.5 line-clamp-2 text-[12px] leading-relaxed text-muted-foreground">{def.description}</p>}
   625→
   626→                    <div className="mt-3 space-y-1">
   627→                      <KeyValue label="Schedule" value={scheduleLabel(def.scheduleStrategy, def.weekdaysCsv, def.specificDate)} />
   628→                      <KeyValue label="Window" value={`${def.serviceStartLocal} – ${def.serviceEndLocal}`} />
   629→                      <KeyValue label="Cutoff" value={`${def.cutoffStrategy === "SAME_DAY" ? "same day" : def.cutoffStrategy === "PREVIOUS_DAY" ? "previous day" : `−${def.cutoffOffsetDays ?? 0} d`} · ${def.cutoffLocalTime}`} />
   630→                      <KeyValue
   631→                        label="Price"
   632→                        value={def.pricingStrategy === "FIXED" ? <Money minor={def.fixedPriceMinor ?? 0} /> : "From formula"}
   633→                      />
   634→                      <KeyValue label="Default" value={`${def.defaultState === "ON" ? "Opted in" : "Opted out"}${def.defaultVisible ? "" : " · hidden"}`} />
   635→                    </div>
   636→
   637→                    {def.deleteRequestedAt && (
   638→                      <p className="mt-2.5 rounded-md bg-danger/10 px-3 py-2 text-[11px] font-medium text-danger">
   639→                        Deletion queued — {fmtDate(def.deleteRequestedAt)}
   640→                      </p>
   641→                    )}
   642→
   643→                    <div className="mt-3.5 flex items-center justify-end gap-2">
   644→                      <ViewButton
   645→                        label="View / Edit"
   646→                        onClick={() => {
   647→                          setEditingDef(def);
   648→                          setWizardOpen(true);
   649→                        }}
   650→                      />
   651→                      <OverflowMenu
   652→                        label={`Actions for ${def.name}`}
   653→                        actions={[
   654→                          { key: "history", label: "Version history", icon: <History />, onSelect: () => setDetailId(def.id) },
   655→                          {
   656→                            key: "edit",
   657→                            label: "Edit (new version)",
   658→                            icon: <Settings2 />,
   659→                            onSelect: () => {
   660→                              setEditingDef(def);
   661→                              setWizardOpen(true);
   662→                            },
   663→                            separatorBefore: true,
   664→                          },
   665→                          ...(inactive
   666→                            ? []
   667→                            : [{ key: "archive", label: "Archive", icon: <Archive />, onSelect: () => setArchiveTarget(def), separatorBefore: true }]),
   668→                          ...(def.deleteRequestedAt
   669→                            ? []
   670→                            : [{ key: "delete", label: "Request deletion…", icon: <Trash2 />, onSelect: () => setDeleteTarget(def), destructive: true }]),
   671→                        ]}
   672→                      />
   673→                    </div>
   674→                  </GlassCard>
   675→                </motion.div>
   676→              );
   677→            })}
   678→          </AnimatePresence>
   679→        </div>
   680→      )}
   681→
   682→      {/* wizard */}
   683→      {wizardOpen && (
   684→        <DefinitionWizard
   685→          open
   686→          onOpenChange={(open) => {
   687→            if (!open) {
   688→              setWizardOpen(false);
   689→              setEditingDef(null);
   690→            }
   691→          }}
   692→          editing={editingDef}
   693→          onSaved={() => invalidate([DEFS_PATH, detailId ? `${DEFS_PATH}/${detailId}` : DEFS_PATH])}
   694→        />
   695→      )}
   696→
   697→      {/* version history dialog */}
   698→      {detailId && (
   699→        <DialogShell2
   700→          open
   701→          onOpenChange={(open) => !open && setDetailId(null)}
   702→          title={detail ? `${detail.name} — versions` : "Version history"}
   703→          description="Each edit snapshots the full configuration. Materialized services keep the version they were generated with."
   704→          wide
   705→        >
   706→          {detailQuery.isLoading ? (
   707→            <ListSkeleton rows={3} />
   708→          ) : !detail ? (
   709→            <ErrorState code={detailQuery.error ? (detailQuery.error as ApiClientError).code : undefined} message="This meal definition could not be found." onRetry={undefined} />
   710→          ) : (
   711→            <div className="space-y-2">
   712→              {(detail.versions ?? []).map((v) => (
   713→                <div key={v.id} className="glass-inset rounded-md p-3.5">
   714→                  <div className="flex items-center justify-between gap-2">
   715→                    <p className="text-sm font-semibold">Version {v.version}</p>
   716→                    <span className="kpi-num text-[11px] text-muted-foreground">{fmtDateTime(v.createdAt, tz)}</span>
   717→                  </div>
   718→                  {v.configSnapshot && (
   719→                    <div className="mt-1.5 grid grid-cols-2 gap-x-4 text-[12px] text-muted-foreground">
   720→                      <span>Name: {String((v.configSnapshot as Record<string, unknown>).name ?? "—")}</span>
   721→                      <span>Window: {String((v.configSnapshot as Record<string, unknown>).serviceStartLocal ?? "—")} – {String((v.configSnapshot as Record<string, unknown>).serviceEndLocal ?? "—")}</span>
   722→                      <span>Cutoff: {String((v.configSnapshot as Record<string, unknown>).cutoffLocalTime ?? "—")}</span>
   723→                      <span>Pricing: {String((v.configSnapshot as Record<string, unknown>).pricingStrategy ?? "—")}</span>
   724→                    </div>
   725→                  )}
   726→                </div>
   727→              ))}
   728→              {(detail.versions ?? []).length === 0 && <p className="text-[13px] text-muted-foreground">No versions recorded yet.</p>}
   729→            </div>
   730→          )}
   731→        </DialogShell2>
   732→      )}
   733→
   734→      {/* archive confirm */}
   735→      {archiveTarget && (
   736→        <ConfirmDialog
   737→          open
   738→          onOpenChange={(open) => !open && setArchiveTarget(null)}
   739→          title={`Archive ${archiveTarget.name}`}
   740→          description="Future services stop being generated. Residents' existing meals and all history stay intact. This can be reversed by editing the meal."
   741→          confirmLabel="Archive meal"
   742→          tone="destructive"
   743→          loading={acting}
   744→          onConfirm={() => void archive(archiveTarget)}
   745→        />
   746→      )}
   747→
   748→      {/* deletion queue */}
   749→      {deleteTarget && (
   750→        <ConfirmDialog
   751→          open
   752→          onOpenChange={(open) => !open && setDeleteTarget(null)}
   753→          title={`Request deletion — ${deleteTarget.name}`}
   754→          description="This queues the definition for deletion after a 30-day safety window, during which the request can be reviewed in the audit trail. The full version history is preserved."
   755→          confirmLabel="Queue deletion"
   756→          tone="destructive"
   757→          requireReason
   758→          reasonPlaceholder="Why is this meal being removed? (required)"
   759→          loading={acting}
   760→          onConfirm={(reason) => void requestDeletion(deleteTarget, reason)}
   761→        />
   762→      )}
   763→    </div>
   764→  );
   765→}
