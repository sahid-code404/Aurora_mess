    50→interface PolicyInfo {

    53→  title: string;

    56→
    57→function normalizePolicies(data: unknown): PolicyInfo[] {
    58→  const obj = data as { items?: unknown } | null;
    59→  const raw = Array.isArray(data)
    60→    ? data
    61→    : Array.isArray(obj?.items)
    62→      ? (obj.items as unknown[])
    63→      : [];
    64→  const out: PolicyInfo[] = [];
    65→  for (const p of raw) {
    66→    if (!p || typeof p !== "object") continue;
    67→    const o = p as Record<string, unknown>;
    68→    const policyId = [o.policyId, o.id].find((v): v is string => typeof v === "string");
    69→    const policyVersionId = [o.policyVersionId, o.versionId, o.id].find(
    70→      (v): v is string => typeof v === "string"
    71→    );
    72→    if (!policyId || !policyVersionId) continue;
    73→    out.push({
    74→      policyId,
    75→      policyVersionId,
    76→      title:
    77→        [o.title, o.name].find((v): v is string => typeof v === "string") ??
    78→        "Community policy",
    79→      version:
    80→        typeof o.version === "string" || typeof o.version === "number"
    81→          ? String(o.version)
    82→          : undefined,
    83→    });
    84→  }
    85→  return out;
    86→}

   384→function SignInForm({ onSuccess }: { onSuccess: () => void }) {
   385→  const [email, setEmail] = useState("");
   386→  const [password, setPassword] = useState("");
   387→  const [pending, setPending] = useState(false);
   388→  const [error, setError] = useState<string | null>(null);
   389→
   390→  async function submit(e: React.FormEvent) {
   391→    e.preventDefault();
   392→    if (pending) return;
   393→    setPending(true);
   394→    setError(null);
   395→    try {
   396→      const data = await api<{ sessionToken?: string }>("/api/v1/auth/login", {
   397→        method: "POST",
   398→        json: { email: email.trim(), password },
   399→      });
   400→      // Cookie-blocked contexts (preview iframe) rely on the Bearer fallback:
   401→      // keep the raw token so subsequent /auth/me calls can authenticate.
   402→      if (data?.sessionToken) {
   403→        persistSessionToken(data.sessionToken);
   404→      }
   405→      onSuccess();
   406→    } catch (err) {
   407→      if (err instanceof ApiClientError) {
   408→        setError(SIGNIN_ERROR_COPY[err.code] ?? err.message);
   409→      } else {
   410→        setError(SIGNIN_ERROR_COPY.NETWORK);
   411→      }
   412→    } finally {
   413→      setPending(false);
   414→    }
   415→  }
   416→
   417→  return (
   418→    <form onSubmit={submit} className="space-y-4.5" noValidate>
   419→      <motion.div
   420→        initial={{ opacity: 0, y: 14 }}
   421→        animate={{ opacity: 1, y: 0 }}

   427→            type="email"
   428→            inputMode="email"

   486→function passwordProblemText(pw: string): string | null {
   487→  const problems: string[] = [];
   488→  if (pw.length < 10) problems.push("at least 10 characters");
   489→  if (!/[a-z]/.test(pw)) problems.push("a lowercase letter");
   490→  if (!/[A-Z]/.test(pw)) problems.push("an uppercase letter");
   491→  if (!/[0-9]/.test(pw)) problems.push("a number");
   492→  return problems.length ? `Password needs ${problems.join(", ")}.` : null;
   493→}

   495→function RegisterForm({ onDone }: { onDone: () => void }) {
   496→  const [form, setForm] = useState<RegisterForm>(EMPTY_REGISTER);
   497→  const [fieldErrors, setFieldErrors] = useState<Partial<Record<keyof RegisterForm, string>>>({});
   498→  const [accepted, setAccepted] = useState<Record<string, boolean>>({});
   499→  const [fallbackAccepted, setFallbackAccepted] = useState(false);
   500→  const [pending, setPending] = useState(false);
   501→  const [error, setError] = useState<string | null>(null);
   502→  const [submitted, setSubmitted] = useState(false);
   503→
   504→  const policiesQuery = useApiQuery<unknown>("/api/v1/auth/policies");
   505→  const policies = useMemo(
   506→    () => (policiesQuery.data ? normalizePolicies(policiesQuery.data) : []),
   507→    [policiesQuery.data]
   508→  );
   509→  const policiesUnavailable =
   510→    policiesQuery.isError || (policiesQuery.isSuccess && policies.length === 0);
   511→
   512→  const set = (key: keyof RegisterForm) => (v: string) =>
   513→    setForm((f) => ({ ...f, [key]: v }));
   514→
   515→  function validate(): boolean {
   516→    const errs: Partial<Record<keyof RegisterForm, string>> = {};
   517→    if (form.fullName.trim().length < 2) errs.fullName = "Enter your full name.";
   518→    if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(form.email.trim()))
   519→      errs.email = "Enter a valid email address.";
   520→    if (form.phone.trim() !== "" && !/^[+0-9 ()-]{6,18}$/.test(form.phone.trim()))
   521→      errs.phone = "Enter a valid phone number.";
   522→    const pwProblem = passwordProblemText(form.password);
   523→    if (pwProblem) errs.password = pwProblem;
   524→    if (form.confirm !== form.password) errs.confirm = "Passwords don't match.";
   525→    if (!policiesUnavailable && policies.some((p) => !accepted[p.policyId])) {
   526→      setError("Please accept the community policies to continue.");
   527→    } else if (policiesUnavailable && !fallbackAccepted) {
   528→      setError("Please confirm you agree to the mess rules to continue.");
   529→    }
   530→    setFieldErrors(errs);
   531→    return Object.keys(errs).length === 0;
   532→  }
   533→
   534→  async function submit(e: React.FormEvent) {
   535→    e.preventDefault();
   536→    if (pending) return;
   537→    setError(null);
   538→    if (!validate()) return;
   539→    setPending(true);
   540→    try {
   541→      await api("/api/v1/auth/register", {
   542→        method: "POST",
   543→        json: {
   544→          fullName: form.fullName.trim(),
   545→          email: form.email.trim().toLowerCase(),
   546→          phone: form.phone.trim() || undefined,
   547→          room: form.room.trim() || undefined,
   548→          password: form.password,
   549→          acceptances: policies.map((p) => ({
   550→            policyId: p.policyId,
   551→            policyVersionId: p.policyVersionId,
   552→          })),
   553→        },
   554→      });
   555→      setSubmitted(true);
   556→    } catch (err) {
   557→      if (err instanceof ApiClientError) {
   558→        if (err.fields) {
   559→          const errs: Partial<Record<keyof RegisterForm, string>> = {};
   560→          for (const [k, v] of Object.entries(err.fields)) {
   561→            const key = k as keyof RegisterForm;
   562→            if (key in EMPTY_REGISTER) errs[key] = v;
   563→          }
   564→          setFieldErrors((prev) => ({ ...prev, ...errs }));
   565→          if (Object.keys(errs).length === 0) setError(err.message);
   566→        } else {
   567→          setError(err.message);
   568→        }
   569→      } else {
   570→        setError(SIGNIN_ERROR_COPY.NETWORK);
   571→      }
   572→    } finally {
   573→      setPending(false);
   574→    }
   575→  }
   576→
   577→  if (submitted) {
   578→    return (
   579→      <motion.div
   580→        initial={{ opacity: 0, scale: 0.94, y: 14 }}
   581→        animate={{ opacity: 1, scale: 1, y: 0 }}
   582→        transition={SPRING_SOFT}
   583→        className="flex flex-col items-center py-6 text-center"
   584→      >
   585→        <motion.span
   586→          initial={{ scale: 0, rotate: -30 }}
   587→          animate={{ scale: 1, rotate: 0 }}
   588→          transition={{ type: "spring", stiffness: 400, damping: 16, delay: 0.15 }}
   589→          className="flex size-16 items-center justify-center rounded-pill border border-success/35 bg-success/14 text-success shadow-[0_10px_30px_-10px_color-mix(in_oklab,var(--success)_60%,transparent)] [&_svg]:size-7"
   590→        >
   591→          <CheckCircle2 aria-hidden />
   592→        </motion.span>
   593→        <h3 className="font-display mt-5 text-lg font-bold">Application submitted</h3>
   594→        <p className="mt-2 max-w-xs text-[13px] leading-relaxed text-muted-foreground">
   595→          Waiting for approval — the admin will review your application. You&apos;ll be
   596→          able to sign in once it&apos;s approved.
   597→        </p>
   598→        <GlassButton variant="secondary" className="mt-6" onClick={onDone}>
   599→          Back to sign in
   600→        </GlassButton>
   601→      </motion.div>
   602→    );
   603→  }
   604→
   605→  return (
   606→    <form onSubmit={submit} className="space-y-4.5" noValidate>
   607→      <div className="grid grid-cols-1 gap-4.5 min-[480px]:grid-cols-2">
   608→        <Field label="Full name" htmlFor="reg-name" error={fieldErrors.fullName}>
   609→          <input
   610→            id="reg-name"
   611→            type="text"
   612→            autoComplete="name"
   613→            value={form.fullName}
   614→            onChange={(e) => set("fullName")(e.target.value)}
   615→            placeholder="Sahid Khan"
   616→            className={inputClass}
   617→          />
   618→        </Field>
   619→        <Field label="Room (optional)" htmlFor="reg-room">
   620→          <input
   621→            id="reg-room"
   622→            type="text"
   623→            value={form.room}
   624→            onChange={(e) => set("room")(e.target.value)}
   625→            placeholder="B-204"
   626→            className={inputClass}
   627→          />
   628→        </Field>
   629→      </div>
   630→      <Field label="Email" htmlFor="reg-email" error={fieldErrors.email}>
   631→        <input
   632→          id="reg-email"
   633→          type="email"
   634→          inputMode="email"
   635→          autoComplete="email"
   636→          value={form.email}
   637→          onChange={(e) => set("email")(e.target.value)}
   638→          placeholder="you@example.com"
   639→          className={inputClass}
   640→        />
   641→      </Field>
   642→      <Field label="Phone (optional)" htmlFor="reg-phone" error={fieldErrors.phone}>
   643→        <input
   644→          id="reg-phone"
   645→          type="tel"
   646→          autoComplete="tel"
   647→          value={form.phone}
   648→          onChange={(e) => set("phone")(e.target.value)}
   649→          placeholder="+91 98765 43210"
   650→          className={inputClass}
   651→        />
   652→      </Field>
   653→      <Field label="Password" htmlFor="reg-password" error={fieldErrors.password}>
   654→        <PasswordInput
   655→          id="reg-password"
   656→          value={form.password}
   657→          onChange={set("password")}
   658→          autoComplete="new-password"
   659→          error={!!fieldErrors.password}
   660→        />
   661→      </Field>
   662→      <Field label="Confirm password" htmlFor="reg-confirm" error={fieldErrors.confirm}>
   663→        <PasswordInput
   664→          id="reg-confirm"
   665→          value={form.confirm}
   666→          onChange={set("confirm")}
   667→          autoComplete="new-password"
   668→          error={!!fieldErrors.confirm}
   669→        />
   670→      </Field>
   671→
   672→      {/* policy acceptance */}
   673→      <div className="glass-inset liquid-sheen space-y-2.5 rounded-xl p-4">
   674→        <p className="text-xs font-semibold text-muted-foreground">Policies</p>
   675→        {policiesQuery.isLoading && <div className="glass-skeleton h-4 w-3/4" />}
   676→        {policiesQuery.isError && (
   677→          <p className="text-[13px] leading-relaxed text-warning">
   678→            We couldn&apos;t load the community policies right now. You can still apply —
   679→            the admin reviews every application.
   680→          </p>
   681→        )}
   682→        {policiesUnavailable && !policiesQuery.isError && (
   683→          <p className="text-[13px] leading-relaxed text-warning">
   684→            No published policies yet. A general acknowledgement is collected instead.
   685→          </p>
   686→        )}
   687→        {policies.map((p) => (
   688→          <label key={p.policyId} className="flex cursor-pointer items-start gap-2.5 text-[13px]">
   689→            <input
   690→              type="checkbox"
   691→              checked={!!accepted[p.policyId]}
   692→              onChange={(e) =>
   693→                setAccepted((a) => ({ ...a, [p.policyId]: e.target.checked }))
   694→              }
   695→              className="mt-0.5 size-4 shrink-0 accent-[var(--primary)]"
   696→            />
   697→            <span className="leading-snug">
   698→              I accept the {p.title}
   699→              {p.version ? (
   700→                <span className="ml-1 text-[11px] text-muted-foreground">v{p.version}</span>
   701→              ) : null}
   702→              .
   703→            </span>
   704→          </label>
   705→        ))}
   706→        {policiesUnavailable && (
   707→          <label className="flex cursor-pointer items-start gap-2.5 text-[13px]">
   708→            <input
   709→              type="checkbox"
   710→              checked={fallbackAccepted}
   711→              onChange={(e) => setFallbackAccepted(e.target.checked)}
   712→              className="mt-0.5 size-4 shrink-0 accent-[var(--primary)]"
   713→            />
   714→            <span className="leading-snug">
   715→              I agree to follow the mess rules and payment policy.
   716→            </span>
   717→          </label>
   718→        )}
   719→      </div>
   720→
   721→      <AnimatePresence>{error && <ErrorSurface message={error} />}</AnimatePresence>
   722→      <GlassButton type="submit" size="lg" fullWidth loading={pending}>
   723→        Submit application
   724→      </GlassButton>
   725→    </form>
   726→  );
   727→}

   775→function SeedFillButton({ email, password }: { email: string; password: string }) {

   777→  const reduced = useReducedMotion();
   778→

   781→    const pwInput = document.getElementById("signin-password") as HTMLInputElement | null;
   782→    if (emailInput) {

   788→      pwInput.dispatchEvent(new Event("input", { bubbles: true }));

   795→    <GlassButton
   796→      variant="ghost"
   797→      size="sm"

   804→  );
   805→}
