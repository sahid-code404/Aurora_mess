    76→function patchCacheRows(
    77→  value: MealsCacheValue | undefined,
    78→  instanceId: string,
    79→  patch: (m: MealInstanceDto) => MealInstanceDto
    80→): MealsCacheValue | undefined {
    81→  if (!value) return value;

    87→/** (year, monthIndex 0-11) → "YYYY-MM". */
    88→function monthKeyOf(year: number, month: number): string {
    89→  return `${year}-${String(month + 1).padStart(2, "0")}`;
    90→}
    91→
    92→function monthStartKey(year: number, month: number): string {
    93→  return `${monthKeyOf(year, month)}-01`;
    94→}
    95→
    96→function monthEndKey(year: number, month: number): string {
    97→  const lastDay = new Date(Date.UTC(year, month + 1, 0)).getUTCDate();
    98→  return `${monthKeyOf(year, month)}-${String(lastDay).padStart(2, "0")}`;
    99→}

   101→/** "September 2025". */
   102→function monthLongLabel(year: number, month: number): string {
   103→  return new Intl.DateTimeFormat("en-IN", { month: "long", year: "numeric", timeZone: "UTC" }).format(
   104→    new Date(Date.UTC(year, month, 1))
   105→  );
   106→}

   108→/** "Sep 2025" — compact KPI sub for non-current months. */
   109→function monthShortKey(year: number, month: number): string {
   110→  return new Intl.DateTimeFormat("en-IN", { month: "short", year: "numeric", timeZone: "UTC" }).format(
   111→    new Date(Date.UTC(year, month, 1))
   112→  );
   113→}

   115→/** Picker pill labels — relative for ±1 day, compact otherwise (BoardOps pattern). */
   116→function dayPillLabels(key: string, todayKey: string): { top: string; bottom: string } {
   117→  if (key === todayKey) return { top: "Today", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   118→  if (key === addDaysToKey(todayKey, -1)) return { top: "Yesterday", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   119→  if (key === addDaysToKey(todayKey, 1)) return { top: "Tomorrow", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   120→  return { top: formatDateInTz(key, "UTC"), bottom: weekdayShort(key) };
   121→}

   241→  /* --------------------------- optimistic toggle --------------------------- */
   242→
   243→  async function handleToggle(meal: MealInstanceDto, next: "ON" | "OFF") {
   244→    const queries = queryClient.getQueriesData<MealsCacheValue>({
   245→      queryKey: ["api", "/api/v1/meals"],
   246→    });
   247→    const snapshots = queries.map(([key, value]) => ({ key, value }));
   248→
   249→    let current: MealInstanceDto | undefined;
   250→    for (const { value } of snapshots) {
   251→      current = value?.data.find((m) => m.id === meal.id);
   252→      if (current) break;
   253→    }
   254→    if (!current) return;
   255→
   256→    const expectedVersion = current.myState.version;
   257→
   258→    // Optimistically flip every cached copy of this row.
   259→    for (const { key } of snapshots) {
   260→      queryClient.setQueryData<MealsCacheValue>(key, (old) =>
   261→        patchCacheRows(old, meal.id, (m) => ({
   262→          ...m,
   263→          myState: { ...m.myState, effectiveState: next } as MealMyState,
   264→        }))
   265→      );
   266→    }
   267→
   268→    try {
   269→      const res = await apiJson<ToggleResponse>(`/api/v1/meals/${meal.id}/toggle`, "POST", {
   270→        state: next,
   271→        expectedVersion,
   272→      });
   273→      // Authoritative response wins.
   274→      for (const { key } of snapshots) {
   275→        queryClient.setQueryData<MealsCacheValue>(key, (old) =>
   276→          patchCacheRows(old, meal.id, (m) => ({
   277→            ...m,
   278→            myState: {
   279→              ...m.myState,
   280→              effectiveState: res.state,
   281→              effectiveReason: res.effectiveReason,
   282→              locked: res.locked,
   283→              version: res.version,
   284→            } as MealMyState,
   285→          }))
   286→        );
   287→      }
   288→      invalidate([RESIDENT_KEYS.dashboard]);
   289→    } catch (err) {
   290→      // Rollback — the toggle springs back (spec §114).
   291→      for (const { key, value } of snapshots) {
   292→        queryClient.setQueryData<MealsCacheValue>(key, value);
   293→      }
   294→      if (err instanceof ApiClientError) {
   295→        if (err.code === "MEAL_CUTOFF_PASSED") {
   296→          flashFor(meal.id, {
   297→            tone: "warning",
   298→            text: `${meal.name} is already locked. Admin can still override it.`,
   299→          });
   300→        } else if (err.code === "RESOURCE_CHANGED") {
   301→          flashFor(meal.id, { tone: "info", text: "This meal was just changed. Refreshing…" });
   302→          void queryClient.invalidateQueries({ queryKey: ["api", "/api/v1/meals"] });
   303→        } else if (err.code === "MEAL_NOT_AVAILABLE") {
   304→          flashFor(meal.id, { tone: "warning", text: "This meal isn't available to change right now." });
   305→        } else {
   306→          flashFor(meal.id, { tone: "danger", text: err.message });
   307→        }
   308→      } else {
   309→        flashFor(meal.id, { tone: "danger", text: friendlyError(err) });
   310→      }
   311→    }
   312→  }

   366→  /**
   367→   * Adjust ONE MEAL's guest count with its row stepper — self-service under
   368→   * cutoff, no admin permission (guest meals behave like normal meals in the
   369→   * user flow). Optimistic patch of every cached guest list copy with rollback
   370→   * and a plain-language flash, exactly like the meal toggles (spec §114).
   371→   * "+1" with no request for that meal opens the guest dialog pre-selected
   372→   * to the meal (a quantity note may be needed).
   373→   */
   374→  async function handleGuestStep(row: AgendaMealRowVm, delta: 1 | -1) {
   375→    const inst = meals.find((m) => m.id === row.id);
   376→    if (!inst) return;
   377→    const dayGuests = guestsByDate.get(inst.serviceDate) ?? [];
   378→    const instanceGuests = dayGuests.filter((g) => g.mealInstanceId === inst.id);
   379→    const queries = queryClient.getQueriesData<{ data: GuestMealDto[]; meta: unknown }>({
   380→      queryKey: ["api", "/api/v1/guest-meals"],
   381→    });
   382→    const snapshots = queries.map(([key, value]) => ({ key, value }));
   383→
   384→    // Optimistic: apply the same target pick locally (this meal's requests only).
   385→    const target = pickGuestStepTarget(instanceGuests, delta, serverNow);
   386→    if (target) {
   387→      const nextQty = delta === 1 ? target.quantity + 1 : target.quantity - 1;
   388→      for (const { key } of snapshots) {
   389→        queryClient.setQueryData<{ data: GuestMealDto[]; meta: unknown }>(key, (old) =>
   390→          old
   391→            ? {
   392→                ...old,
   393→                data: old.data.map((g) =>
   394→                  g.id !== target.id
   395→                    ? g
   396→                    : nextQty <= 0
   397→                      ? { ...g, status: "CANCELLED" }
   398→                      : { ...g, quantity: nextQty, totalPriceMinor: g.unitPriceMinor * nextQty }
   399→                ),
   400→              }
   401→            : old
   402→        );
   403→      }
   404→    }
   405→
   406→    const result = await stepGuestMeals(instanceGuests, delta, serverNow);
   407→    if (result.kind === "ok") {
   408→      invalidate([RESIDENT_KEYS.guestMeals, RESIDENT_KEYS.dashboard, RESIDENT_KEYS.billing, RESIDENT_KEYS.notifications]);
   409→    } else if (result.kind === "dialog") {
   410→      // No request for this meal can absorb the +1 — the dialog takes over,
   411→      // pre-selected to the meal the user tapped.
   412→      for (const { key, value } of snapshots) queryClient.setQueryData(key, value);
   413→      openGuestDialog(inst.serviceDate, inst.id);
   414→    } else {
   415→      // Rollback — the counter springs back.
   416→      for (const { key, value } of snapshots) queryClient.setQueryData(key, value);
   417→      if (result.code === "RESOURCE_CHANGED") {
   418→        flashGuestFor(inst.id, { tone: "info", text: "This guest meal was just changed. Refreshing…" });
   419→        void queryClient.invalidateQueries({ queryKey: ["api", "/api/v1/guest-meals"] });
   420→      } else if (result.code === "MEAL_CUTOFF_PASSED") {
   421→        flashGuestFor(inst.id, {
   422→          tone: "warning",
   423→          text: `${inst.name} is locked — the cutoff already passed.`,
   424→        });
   425→      } else {
   426→        flashGuestFor(inst.id, { tone: "danger", text: result.message });
   427→      }
   428→    }
   429→  }
