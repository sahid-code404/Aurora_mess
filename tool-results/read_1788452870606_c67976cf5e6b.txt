    53→interface DraftItem {
    54→  key: string;
    55→  itemName: string;
    56→  quantity: string;
    57→  unit: string;
    58→  unitPrice: string;
    59→}
    60→
    61→function draftItem(): DraftItem {
    62→  return { key: crypto.randomUUID(), itemName: "", quantity: "1", unit: "kg", unitPrice: "" };
    63→}
    64→
    65→function itemEstimateMinor(item: DraftItem): number | null {
    66→  const price = parseMoneyToMinor(item.unitPrice);
    67→  if (price == null) return null;
    68→  const qty = Number(item.quantity);
    69→  if (!Number.isFinite(qty) || qty <= 0) return null;
    70→  // Same rounding as the server: round-half-up on each line.
    71→  return Math.round(qty * price);
    72→}

   391→/** "2025-09" ± 1 → "2025-08" / "2025-10". */
   392→function shiftMonthKey(key: string, delta: number): string {
   393→  const [y, m] = key.split("-").map(Number);
   394→  const d = new Date(y ?? 2025, (m ?? 1) - 1 + delta, 1);
   395→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
   396→}
   397→
   398→/** Long month name for the picker pill ("September"). */
   399→function monthLongName(key: string): string {
   400→  const [y, m] = key.split("-").map(Number);
   401→  return new Intl.DateTimeFormat("en-US", { month: "long" }).format(new Date(y ?? 2025, (m ?? 1) - 1, 1));
   402→}
   403→
   404→/** "Sep 2025" for the month-scoped KPI label. */
   405→function monthShortLabel(key: string): string {
   406→  return monthLabel(Number(key.slice(0, 4)), Number(key.slice(5, 7)));
   407→}

   409→export default function AdminExpenses() {
   410→  const [status, setStatus] = useState("ALL");
   411→  // Month scope — undefined = the server's current month (institution tz).
   412→  // Navigating sets an explicit YYYY-MM; the list AND the month KPIs follow it
   413→  // (the route scopes both to the requested month and echoes it in meta.month).
   414→  const [monthParam, setMonthParam] = useState<string | undefined>(undefined);
   415→  const [formOpen, setFormOpen] = useState(false);
   416→  const [detail, setDetail] = useState<ExpenseRow | null>(null);
   417→  const [action, setAction] = useState<"approve" | "reject" | "void" | null>(null);
   418→  const [acting, setActing] = useState(false);
   419→  const invalidate = useInvalidate();
   420→  const { institution } = useSession();
   421→  const tz = institution?.timezone ?? "Asia/Kolkata";
   422→
   423→  const { data: envelope, isLoading, error, refetch } = useApiMetaQuery<ExpenseRow[]>(EXPENSES_PATH, {
   424→    status: status === "ALL" ? undefined : status,
   425→    month: monthParam,
   426→  });
   427→  const expenses = envelope?.data ?? [];
   428→  const meta = envelope?.meta ?? {};
   429→
   430→  const categoriesQuery = useApiQuery<ExpenseCategory[]>(formOpen ? CATEGORIES_PATH : null);
   431→  const categories = categoriesQuery.data ?? [];
   432→
   433→  const pendingCount = expenses.filter((e) => e.status === "PENDING").length;
   434→
   435→  // Effective month — the server echoes the resolved month in meta (it owns
   436→  // the institution timezone); fall back to the client month before load.
   437→  const thisMonthKey = todayKey().slice(0, 7);
   438→  const activeMonthKey = metaStr(meta, "month") ?? monthParam ?? thisMonthKey;
   439→  const isThisMonth = activeMonthKey === thisMonthKey;
   440→
   441→  async function runAction(kind: "approve" | "reject" | "void", reason?: string) {
   442→    if (!detail) return;
   443→    setActing(true);
   444→    try {
   445→      await postJson(`${EXPENSES_PATH}/${detail.id}/${kind}`, kind === "approve" ? {} : { reason });
   446→      invalidate([EXPENSES_PATH, "/api/v1/admin/funds", "/api/v1/admin/dashboard", "/api/v1/admin/billing"]);
   447→      toast.success(kind === "approve" ? "Expense approved" : kind === "reject" ? "Expense rejected" : "Expense voided", {
   448→        description: `${detail.displayNumber} · ${detail.totalFormatted}`,
   449→      });
   450→      setAction(null);
   451→      setDetail(null);
   452→    } catch (err) {
   453→      toast.error(errMessage(err));
   454→    } finally {
   455→      setActing(false);
   456→    }
   457→  }
   458→
   459→  if (error) {
   460→    return (
   461→      <div className="space-y-4">
   462→        <ErrorState
   463→          code={(error as ApiClientError | undefined)?.code}
   464→          message={(error as ApiClientError | undefined)?.message}
   465→          onRetry={() => void refetch()}
   466→        />
   467→      </div>
   468→    );
   469→  }
   470→
   471→  return (
   472→    <StaggerGroup className="space-y-4">
   473→      {/* Month capsule — circular arrows + reset pill (BoardOps picker) */}
   474→      <StaggerItem>
   475→      <PickerCapsule
   476→        onPrev={() => setMonthParam(shiftMonthKey(activeMonthKey, -1))}
   477→        onNext={() => setMonthParam(shiftMonthKey(activeMonthKey, 1))}
   478→        prevLabel="Previous month"
   479→        nextLabel="Next month"
   480→        onPillClick={() => setMonthParam(undefined)}
   481→        pillAriaLabel="Reset to the current month"
   482→        resettable={!isThisMonth}
   483→      >
   484→        <Calendar className="size-4 shrink-0 text-primary" aria-hidden />
   485→        <span className="min-w-0 text-center leading-tight">
   486→          <span className="block truncate text-sm font-bold text-primary">{monthLongName(activeMonthKey)}</span>
   487→          <span className="block truncate text-[11px] text-muted-foreground">{activeMonthKey.slice(0, 4)}</span>
   488→        </span>
   489→      </PickerCapsule>
   490→      </StaggerItem>
   491→
   492→      <StaggerItem>
   493→      <KpiGrid
   494→        loading={isLoading && !envelope}
   495→        className="lg:grid-cols-3"
   496→        kpis={[
   497→          {
   498→            label: `Total Expenses · ${monthShortLabel(activeMonthKey)}`,
   499→            value: metaStr(meta, "expensesThisMonthFormatted") ?? "—",
   500→            icon: <ReceiptText />,
   501→            tone: "primary",
   502→            glow: "primary",
   503→            sub: `${metaNum(meta, "entriesThisMonth") ?? 0} item entries approved`,
   504→          },
   505→          {
   506→            label: "Remaining Funds",
   507→            value: metaStr(meta, "remainingFundsFormatted") ?? "—",
   508→            icon: <Package />,
   509→            tone: "success",
   510→            glow: "success",
   511→            sub: "Cash & bank balance",
   512→          },
   513→          {
   514→            label: "Awaiting Review",
   515→            value: String(pendingCount),
   516→            icon: <FileText />,
   517→            tone: "warning",
   518→            glow: "warning",
   519→            sub: "Pending expenses",
   520→          },
   521→        ]}
   522→      />
   523→      </StaggerItem>
   524→
   525→      <StaggerItem>
   526→      <FilterChips
   527→        chips={[
   528→          { value: "ALL", label: "All" },
   529→          { value: "PENDING", label: "Pending" },
   530→          { value: "APPROVED", label: "Approved" },
   531→          { value: "REJECTED", label: "Rejected" },
   532→          { value: "VOIDED", label: "Voided" },
   533→        ]}
   534→        value={status}
   535→        onChange={setStatus}
   536→      />
   537→      </StaggerItem>
   538→
   539→      <StaggerItem>
   540→      {/* Primary action — BoardOps action bar, right-aligned above the list */}
   541→      <div className="flex flex-wrap items-center justify-end gap-2">
   542→        <GlassButton variant="primary" icon={<Plus />} onClick={() => setFormOpen(true)}>
   543→          Add Expense
   544→        </GlassButton>
   545→      </div>
   546→      </StaggerItem>
   547→
   548→      <StaggerItem>
   549→      {isLoading && !envelope ? (
   550→        <ListSkeleton rows={5} />
   551→      ) : expenses.length === 0 ? (
   552→        <EmptyState
   553→          icon={ReceiptText}
   554→          title={status === "PENDING" ? "No pending expenses" : "No expenses recorded yet"}
   555→          description={
   556→            status === "PENDING"
   557→              ? "Recorded expenses waiting for your approval will appear here."
   558→              : "Use Add Expense to record the first mess purchase."
   559→          }
   560→          action={
   561→            status !== "PENDING" ? (
   562→              <GlassButton variant="secondary" icon={<Plus />} onClick={() => setFormOpen(true)}>
   563→                Add Expense
   564→              </GlassButton>
   565→            ) : undefined
   566→          }
   567→        />
   568→      ) : (
   569→        <div className="max-h-[28rem] space-y-2.5 overflow-y-auto pr-1">
   570→          <AnimatePresence mode="popLayout">
   571→            {expenses.map((e, i) => (
   572→              <motion.div
   573→                key={e.id}
   574→                initial={{ opacity: 0, y: 6 }}
   575→                animate={{ opacity: 1, y: 0 }}
   576→                exit={{ opacity: 0, y: -4, scale: 0.98, transition: { duration: 0.15 } }}
   577→                transition={{ ...SPRING_SNAPPY, delay: Math.min(i * 0.04, 0.2) }}
   578→              >
   579→                <GlassCard className="p-4">
   580→                  <div className="flex items-start gap-3">
   581→                    {(() => {
   582→                      const tile = expenseTile(e.status);
   583→                      const TileIcon = tile.Icon;
   584→                      return (
   585→                        <span
   586→                          aria-hidden
   587→                          className={cn(
   588→                            "flex size-10 shrink-0 items-center justify-center rounded-md border [&_svg]:size-5",
   589→                            tile.className
   590→                          )}
   591→                        >
   592→                          <TileIcon aria-hidden />
   593→                        </span>
   594→                      );
   595→                    })()}
   596→                    <div className="min-w-0 flex-1">
   597→                      <div className="flex flex-wrap items-center gap-2">
   598→                        <p className="min-w-0 truncate text-sm font-medium" title={e.description}>
   599→                          {e.description}
   600→                        </p>
   601→                        <StatusBadge status={e.status} />
   602→                        {e.source === "TASK" && <Chip tone="frost">via task</Chip>}
   603→                      </div>
   604→                      <p className="kpi-num mt-0.5 truncate text-[11px] text-muted-foreground">
   605→                        {e.displayNumber} · {e.categoryName ?? "No category"} · {e.dateKey ? fmtDate(e.dateKey) : "—"} ·{" "}
   606→                        {e.itemCount ?? 0} item{(e.itemCount ?? 0) === 1 ? "" : "s"}
   607→                      </p>
   608→                      <div className="mt-2 flex flex-wrap items-baseline gap-x-4 gap-y-1">
   609→                        <div className="flex items-baseline gap-1">
   610→                          <span className="text-[10px] font-medium uppercase tracking-wide text-muted-foreground">
   611→                            Total
   612→                          </span>
   613→                          <Money minor={e.totalMinor} className="text-base font-bold" />
   614→                        </div>
   615→                        <span className="kpi-num text-[11px] text-muted-foreground/80">
   616→                          Recorded {fmtDateTime(e.createdAt, tz)}
   617→                        </span>
   618→                      </div>
   619→                    </div>
   620→                    <OverflowMenu
   621→                      actions={[
   622→                        { key: "view", label: "View details", icon: <FileText />, onSelect: () => setDetail(e) },
   623→                        ...(e.status === "PENDING"
   624→                          ? [
   625→                              { key: "approve", label: "Approve", icon: <CheckCircle2 />, onSelect: () => setDetail(e), separatorBefore: true },
   626→                              { key: "reject", label: "Reject…", icon: <XCircle />, onSelect: () => setDetail(e) },
   627→                            ]
   628→                          : []),
   629→                        ...(e.status === "APPROVED"
   630→                          ? [{ key: "void", label: "Void…", icon: <Trash2 />, onSelect: () => setDetail(e), destructive: true, separatorBefore: true }]
   631→                          : []),
   632→                      ]}
   633→                    />
   634→                  </div>
   635→                </GlassCard>
   636→              </motion.div>
   637→            ))}
   638→          </AnimatePresence>
   639→        </div>
   640→      )}
   641→      </StaggerItem>
   642→
   643→      <ExpenseFormDialog
   644→        open={formOpen}
   645→        onOpenChange={setFormOpen}
   646→        categories={categories}
   647→        onSaved={() => invalidate([EXPENSES_PATH, "/api/v1/admin/dashboard"])}
   648→      />
   649→
   650→      <ExpenseDetailDialog
   651→        expense={detail}
   652→        tz={tz}
   653→        onClose={() => setDetail(null)}
   654→        onAction={(kind) => setAction(kind)}
   655→      />
   656→
   657→      {action && detail && (
   658→        <ConfirmDialog
   659→          open
   660→          onOpenChange={(open) => !open && setAction(null)}
   661→          title={
   662→            action === "approve"
   663→              ? "Approve expense"
   664→              : action === "reject"
   665→                ? "Reject expense"
   666→                : "Void expense"
   667→          }
   668→          description={
   669→            action === "approve" ? (
   670→              <>
   671→                The money is posted to the mess ledger (Dr Expenses / Cr Cash) and counts toward this month's formula.
   672→                <span className="mt-2 block font-medium">
   673→                  {detail.displayNumber} · {detail.totalFormatted}
   674→                </span>
   675→              </>
   676→            ) : action === "reject" ? (
   677→              <>
   678→                The record stays for the audit trail but no money moves. The submitter is notified with your reason.
   679→                <span className="mt-2 block font-medium">
   680→                  {detail.displayNumber} · {detail.totalFormatted}
   681→                </span>
   682→              </>
   683→            ) : (
   684→              <>
   685→                A reversal journal is posted — the money returns to cash and the expense stays visible as voided. Approved
   686→                expenses are never deleted.
   687→                <span className="mt-2 block font-medium">
   688→                  {detail.displayNumber} · {detail.totalFormatted}
   689→                </span>
   690→              </>
   691→            )
   692→          }
   693→          confirmLabel={action === "approve" ? "Approve" : action === "reject" ? "Reject" : "Void"}
   694→          tone={action === "approve" ? "primary" : "destructive"}
   695→          requireReason={action !== "approve"}
   696→          loading={acting}
   697→          onConfirm={(reason) => void runAction(action, reason)}
   698→        />
   699→      )}
   700→    </StaggerGroup>
   701→  );
   702→}
