   121→   121→function toDateString(d: Date): string {
   122→   122→  const y = d.getFullYear();
   123→   123→  const m = String(d.getMonth() + 1).padStart(2, "0");
   124→   124→  const day = String(d.getDate()).padStart(2, "0");
   125→   125→  return `${y}-${m}-${day}`;
   126→   126→}
   127→   127→
   128→   128→function to12h(t: string): string {
   129→   129→  const [h, m] = t.split(":").map(Number);
   130→   130→  const period = h >= 12 ? "PM" : "AM";
   131→   131→  const hr = h % 12 || 12;
   132→   132→  return `${hr}:${String(m).padStart(2, "0")} ${period}`;
   133→   133→}
   134→   134→
   135→   135→function parseDateStr(s: string): Date {
   136→   136→  const [y, m, d] = s.split("-").map(Number);
   137→   137→  return new Date(y, m - 1, d);
   138→   138→}
