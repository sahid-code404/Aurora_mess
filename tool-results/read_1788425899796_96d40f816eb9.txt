   165→function formatINR(n: number) {
   166→  return `₹${Math.round(n).toLocaleString("en-IN")}`;
   167→}
   168→
   169→function formatMonthYear(month: number, year: number) {
   170→  return `${MONTHS[month] ?? "—"} ${year}`;
   171→}
   172→
   173→function formatDate(iso: string | null) {
   174→  if (!iso) return "—";
   175→  const d = new Date(iso);
   176→  const datePart = d.toLocaleDateString("en-GB", { day: "numeric", month: "short", year: "numeric" });
   177→  const timePart = d.toLocaleTimeString("en-US", { hour: "2-digit", minute: "2-digit", hour12: true });
   178→  return `${datePart}, ${timePart}`;
   179→}
   180→
   181→/** Full date-time format: "29 June 2026, 05:28 am" */
   182→function formatDateTime(iso: string | null) {
   183→  if (!iso) return "—";
   184→  const d = new Date(iso);
   185→  const datePart = d.toLocaleDateString("en-GB", { day: "numeric", month: "long", year: "numeric" });
   186→  const timePart = d.toLocaleTimeString("en-US", { hour: "2-digit", minute: "2-digit", hour12: true });
   187→  return `${datePart}, ${timePart}`;
   188→}
   189→
   190→/** A bill is overdue if it has a due date in the past AND there's still an outstanding due amount. */
   191→function isOverdue(bill: Bill): boolean {
   192→  if (!bill.dueDate || bill.deletedAt) return false;
   193→  if (bill.status === "PAID" || bill.status === "VOID") return false;
   194→  return new Date(bill.dueDate) < new Date() && bill.dueAmount > 0;
   195→}

   242→        params: { month: selectedMonth, year: selectedYear, includeDeleted: "true" },
   243→      });

   245→    },
   246→    enabled: isAdmin,
   247→    placeholderData: (prev) => prev,
   248→  });
   249→
   250→  const voidMutation = useMutation({

   252→      api.delete<ApiResponse<{ success: boolean }>>(`/bills/${id}`),
   253→    onSuccess: () => {
   254→      toast.success("Bill voided successfully");
   255→      setVoidTarget(null);
   256→      qc.invalidateQueries({ queryKey: ["bills"] });
   257→    },
   258→    onError: (e: Error) => toast.error(e.message || "Failed to void bill"),
   259→  });
   260→
   261→  const [deleteTarget, setDeleteTarget] = useState<Bill | null>(null);
   262→  const [deleteAllOpen, setDeleteAllOpen] = useState(false);
   263→  const [deleteReason, setDeleteReason] = useState("");
   264→  const [deleteAllReason, setDeleteAllReason] = useState("");

   266→  const deleteMutation = useMutation({
   267→    mutationFn: async (id: string) => {
   268→      await api.delete(`/bills/${id}`, { body: JSON.stringify({ reason: deleteReason || undefined }) });
   269→    },
   270→    onSuccess: () => {
   271→      toast.success("Bill scheduled for deletion — permanently removed in 7 days");
   272→      setDeleteTarget(null);
   273→      setDeleteReason("");
   274→      qc.invalidateQueries({ queryKey: ["bills"] });
   275→    },
   276→    onError: (e: Error) => toast.error(e.message || "Failed to delete bill"),
   277→  });
   278→
   279→  const deleteAllMutation = useMutation({
   280→    mutationFn: async () => {
   281→      const r = await api.delete<ApiResponse<{ deleted: number }>>("/bills", {
   282→        params: { month: selectedMonth, year: selectedYear },

   286→    },
   287→    onSuccess: (data) => {
   288→      toast.success(`${data.deleted} bills scheduled for deletion — permanently removed in 7 days`);
   289→      setDeleteAllOpen(false);
   290→      setDeleteAllReason("");
   291→      qc.invalidateQueries({ queryKey: ["bills"] });
   292→    },
   293→    onError: (e: Error) => toast.error(e.message || "Failed to delete bills"),
   294→  });
   295→
   296→  const restoreMutation = useMutation({
   297→    mutationFn: async (id: string) => {
   298→      const r = await api.post<ApiResponse<Bill>>(`/bills/${id}/restore`);
   299→      return r.data;
   300→    },
   301→    onSuccess: () => {
   302→      toast.success("Bill restored successfully");
   303→      qc.invalidateQueries({ queryKey: ["bills"] });
   304→    },
   305→    onError: (e: Error) => toast.error(e.message || "Failed to restore bill"),
   306→  });

   308→  const restoreAllMutation = useMutation({
   309→    mutationFn: async () => {
   310→      const results = await Promise.all(
   311→        deletedBills.map((b) => api.post(`/bills/${b.id}/restore`).catch(() => null))

   314→    },
   315→    onSuccess: (count) => {
   316→      toast.success(`${count} bills restored successfully`);
   317→      qc.invalidateQueries({ queryKey: ["bills"] });

   319→    onError: (e: Error) => toast.error(e.message || "Failed to restore bills"),
   320→  });
   321→
   322→  // ── Derived KPIs ──
   323→  // KPIs include ALL bills (active + soft-deleted) for the month, excluding only VOID.
   324→  // Soft-deleting a bill removes it from the user's view but the financial data
   325→  // (total billed, collected, outstanding) still counts in the KPIs.
   326→  const allBills = useMemo(() => [...bills, ...deletedBills], [bills, deletedBills]);
   327→  const kpis = useMemo(() => {
   328→    const active = allBills.filter((b) => b.status !== "VOID");
   329→    const totalBilled = active.reduce((s, b) => s + b.totalAmount, 0);
   330→    const totalCollected = active.reduce((s, b) => s + b.paidAmount, 0);
   331→    const overdueBills = active.filter((b) => isOverdue(b));
   332→    const overdueCount = overdueBills.length;
   333→    const overdueAmount = overdueBills.reduce((s, b) => s + b.dueAmount, 0);
   334→    return { totalBilled, totalCollected, overdueCount, overdueAmount };
   335→  }, [allBills]);
   336→
   337→  // ── Filtered list ──
   338→  // Use deleted bills when filter is DELETED, otherwise use active bills
   339→  const sourceBills = statusFilter === "DELETED" ? deletedBills : bills;
   340→
   341→  const filtered = useMemo(() => {
   342→    const q = search.trim().toLowerCase();
   343→    return sourceBills.filter((b) => {
   344→      if (statusFilter !== "ALL" && statusFilter !== "DELETED" && b.status !== statusFilter) return false;
   345→      if (!q) return true;
   346→      return (
   347→        b.user.name?.toLowerCase().includes(q) ||
   348→        b.user.email?.toLowerCase().includes(q) ||
   349→        (b.user.room || "").toLowerCase().includes(q)
   350→      );
   351→    });
   352→  }, [sourceBills, search, statusFilter]);
   353→

   357→        <div className={cn("gap-3", isAdmin ? "grid grid-cols-3" : "grid grid-cols-2")}>

   359→            <ShimmerSkeleton key={i} className="h-32" />
   360→          ))}
   361→        </div>
   362→        <ShimmerSkeleton className="h-14 w-full" />
   363→        <div className="grid gap-3">
   364→          {Array.from({ length: 5 }).map((_, i) => (
   365→            <ShimmerSkeleton key={i} className="h-20" />
   366→          ))}
   367→        </div>
   368→      </div>
   369→    );
   370→  }
   371→
   372→  return (
   373→    <StaggerGroup className="space-y-4">
   374→      {/* Subtle refetch indicator — thin animated bar at the top of the list.

   376→          NOT on the initial load (the full skeleton handles that). */}

   403→            <ChevronLeft className="h-5 w-5" />

   416→            <Calendar className="h-4 w-4 text-primary shrink-0" />

   467→          </div>

   469→      ) : (

   485→      {/* KPIs — admins see 3 (Total Billed, Total Collected, Overdue), users see 2 (Total Billed, Overdue) */}

   487→        <div className={cn("gap-3", isAdmin ? "grid grid-cols-3" : "grid grid-cols-2")}>

   515→      </StaggerItem>
   516→
   517→      {/* Search + Filter pills — all in a single horizontal scrollable line */}
   518→      <StaggerItem>
   519→        <div className="space-y-3">

   525→          />
   526→          <div className="flex items-center gap-1 overflow-x-auto no-scrollbar pb-1">
   527→            {(
   528→              [
   529→                "ALL",
   530→                "GENERATED",
   531→                "PARTIALLY_PAID",
   532→                "PAID",

   534→                "VOID",
