   228→interface GenerateResult {

   230→  mealChargeMinor: number;
   231→  mealChargeFormatted: string;

   240→function ReadinessPanel({ period }: { period: BillingPeriodRow }) {
   241→  const readinessPath = `${PERIODS_PATH}/${period.id}/readiness`;
   242→  const { data, isLoading, error, refetch } = useApiQuery<ReadinessData>(readinessPath);
   243→  const [answer, setAnswer] = useState("");
   244→  const [generating, setGenerating] = useState(false);
   245→  const [result, setResult] = useState<GenerateResult | null>(null);
   246→  const invalidate = useInvalidate();
   247→
   248→  if (isLoading) {
   249→    return (
   250→      <GlassCard className="space-y-4 p-4 sm:p-6">
   251→        <SectionHeading>Readiness · {period.monthLabel}</SectionHeading>
   252→        <ListSkeleton rows={4} />
   253→      </GlassCard>
   254→    );
   255→  }
   256→
   257→  if (error || !data) {
   258→    return (
   259→      <GlassCard className="p-4 sm:p-6">
   260→        <SectionHeading>Readiness · {period.monthLabel}</SectionHeading>
   261→        <div className="mt-3">
   262→          <ErrorState
   263→            code={(error as ApiClientError | undefined)?.code}
   264→            message={(error as ApiClientError | undefined)?.message}
   265→            onRetry={() => void refetch()}
   266→          />
   267→        </div>
   268→      </GlassCard>
   269→    );
   270→  }
   271→
   272→  const failed = data.checks.filter((c) => !c.pass);
   273→  const mealChargeBroken = /NaN/i.test(data.summary.mealChargeFormatted ?? "");
   274→  const challengeOk = Number(answer) === data.arithmeticChallenge.a + data.arithmeticChallenge.b;
   275→  const canGenerate = data.ready && challengeOk && !mealChargeBroken;
   276→
   277→  async function generate() {
   278→    setGenerating(true);
   279→    try {
   280→      const res = await postJson<GenerateResult>(`${PERIODS_PATH}/${period.id}/generate`, {
   281→        a: data!.arithmeticChallenge.a,
   282→        b: data!.arithmeticChallenge.b,
   283→        answer: Number(answer),
   284→      });
   285→      setResult(res);
   286→      invalidate([PERIODS_PATH, BILLS_PATH, "/api/v1/admin/dashboard", "/api/v1/admin/funds"]);
   287→      toast.success("Bills generated", { description: `${period.monthLabel} · ${res.billCount} bills · ${res.totalBilledFormatted}` });
   288→    } catch (err) {
   289→      toast.error(errMessage(err));
   290→    } finally {
   291→      setGenerating(false);
   292→    }
   293→  }
   294→
   295→  if (result) {
   296→    return (
   297→      <GlassCard className="space-y-4 p-4 sm:p-6">
   298→        <SectionHeading>Billing generated · {period.monthLabel}</SectionHeading>
   299→        <div className="grid grid-cols-2 gap-3 min-[420px]:grid-cols-4">
   300→          <KeyValue stacked label="Bills" value={<span className="kpi-num text-lg font-semibold">{result.billCount}</span>} />
   301→          <KeyValue stacked label="Meal charge" value={<span className="kpi-num text-lg font-semibold">{result.mealChargeFormatted}</span>} />
   302→          <KeyValue stacked label="Total billed" value={<span className="kpi-num text-lg font-semibold">{result.totalBilledFormatted}</span>} />
   303→          <KeyValue stacked label="Total due" value={<span className="kpi-num text-lg font-semibold">{result.totalDueFormatted}</span>} />
   304→        </div>
   305→        <p className="text-[12px] leading-relaxed text-muted-foreground">
   306→          A frozen snapshot of every input was stored with a checksum. Residents have been notified. Corrections happen
   307→          through bill adjustments only.
   308→        </p>
   309→        <div className="flex justify-end">
   310→          <GlassButton variant="secondary" onClick={() => setResult(null)}>
   311→            Back to readiness
   312→          </GlassButton>
   313→        </div>
   314→      </GlassCard>
   315→    );
   316→  }
   317→
   318→  return (
   319→    <GlassCard className="space-y-5 p-4 sm:p-6">
   320→      <SectionHeading>Readiness · {period.monthLabel}</SectionHeading>
   321→
   322→      {/* summary numbers */}
   323→      <div className="grid grid-cols-2 gap-3 min-[420px]:grid-cols-3 lg:grid-cols-6">
   324→        <KeyValue stacked label="Residents" value={<span className="kpi-num text-base font-semibold">{data.summary.residentCount}</span>} />
   325→        <KeyValue stacked label="Resident meals" value={<span className="kpi-num text-base font-semibold">{data.summary.residentMealCount}</span>} />
   326→        <KeyValue stacked label="Guest meals" value={<span className="kpi-num text-base font-semibold">{data.summary.guestMealCount}</span>} />
   327→        <KeyValue stacked label="Expenses" value={<span className="kpi-num text-base font-semibold">{data.summary.eligibleExpensesFormatted}</span>} />
   328→        <KeyValue stacked label="Payments" value={<span className="kpi-num text-base font-semibold">{data.summary.approvedPaymentsFormatted}</span>} />
   329→        <KeyValue
   330→          stacked
   331→          label="Meal charge"
   332→          value={
   333→            <span className="kpi-num text-base font-semibold">
   334→              {mealChargeBroken ? "—" : data.summary.mealChargeFormatted}
   335→            </span>
   336→          }
   337→        />
   338→      </div>
   339→      {mealChargeBroken && (
   340→        <p className="text-[12px] leading-relaxed text-warning">
   341→          The meal charge can't be computed yet — approve expenses so the formula has numbers to work with.
   342→        </p>
   343→      )}
   344→
   345→      {/* checklist */}
   346→      <div>
   347→        <p className="mb-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   348→          Checks {failed.length === 0 ? "— all clear" : `— ${failed.length} blocking`}
   349→        </p>
   350→        <div className="grid grid-cols-1 gap-1.5 md:grid-cols-2">
   351→          {data.checks.map((check) => (
   352→            <div
   353→              key={check.key}
   354→              className={cn(
   355→                "glass-inset flex items-center gap-2.5 rounded-md px-3.5 py-2.5",
   356→                !check.pass && "border border-warning/30"
   357→              )}
   358→            >
   359→              {check.pass ? (
   360→                <CheckCircle2 className="size-4 shrink-0 text-success" aria-hidden />
   361→              ) : (
   362→                <XCircle className="size-4 shrink-0 text-warning" aria-hidden />
   363→              )}
   364→              <p className="min-w-0 flex-1 truncate text-[13px] font-medium" title={check.label}>
   365→                {check.label}
   366→              </p>
   367→            </div>
   368→          ))}
   369→        </div>
   370→      </div>
   371→
   372→      {/* arithmetic confirmation */}
   373→      <div className="glass-inset space-y-3.5 rounded-md p-4">
   374→        <p className="text-[13px] leading-relaxed text-muted-foreground">
   375→          This generates immutable bills for every resident. To confirm you're ready, solve:
   376→        </p>
   377→        <div className="flex items-center justify-center gap-3">
   378→          <span className="kpi-num text-2xl font-semibold">{data.arithmeticChallenge.a}</span>
   379→          <span className="text-2xl font-semibold">+</span>
   380→          <span className="kpi-num text-2xl font-semibold">{data.arithmeticChallenge.b}</span>
   381→          <span className="text-2xl font-semibold">=</span>
   382→          <input
   383→            type="number"
   384→            inputMode="numeric"
   385→            value={answer}
   386→            onChange={(e) => setAnswer(e.target.value)}
   387→            aria-label="Answer"
   388→            className="glass h-11 w-20 rounded-md text-center text-lg font-semibold outline-none focus-visible:outline-2 focus-visible:outline-offset-1 focus-visible:outline-ring"
   389→            placeholder="?"
   390→          />
   391→        </div>
   392→        <div className="flex items-center justify-between gap-3">
   393→          <p className="text-[11px] leading-relaxed text-muted-foreground">
   394→            {data.ready ? "All checks pass — waiting on your confirmation." : "Resolve the failing checks above first."}

   396→          <GlassButton

   398→            icon={<ReceiptText />}

   413→interface PeriodDetailData {

   420→    residentMealCount: number;
   421→    guestMealCount: number;

   423→    approvedPaymentsFormatted: string;

   425→    mealChargeFormatted: string;
   426→    formula: { version: number; expression: string; mealChargeMinor: number } | null;
   427→  } | null;
   428→  bills: (BillRow & { residentName: string })[];

   430→
   431→function BilledPanel({ period }: { period: BillingPeriodRow }) {
   432→  const { data, isLoading, error, refetch } = useApiQuery<PeriodDetailData>(`${PERIODS_PATH}/${period.id}`);
   433→  const [reopenOpen, setReopenOpen] = useState(false);
   434→  const [acting, setActing] = useState(false);
   435→  const invalidate = useInvalidate();
   436→  const { institution } = useSession();
   437→  const tz = institution?.timezone ?? "Asia/Kolkata";
   438→
   439→  // 48h reopen window (server enforces; this is a UX hint)
   440→  const reopenWindowOpen = useMemo(() => {
   441→    if (!period.billedAt) return false;
   442→    return Date.now() - new Date(period.billedAt).getTime() < 48 * 3600 * 1000;
   443→  }, [period.billedAt]);
   444→
   445→  async function reopen(reason: string | undefined) {
   446→    setActing(true);
   447→    try {
   448→      await postJson(`${PERIODS_PATH}/${period.id}/reopen`, { reason });
   449→      invalidate([PERIODS_PATH, BILLS_PATH]);
   450→      toast.success("Period reopened", { description: "Bills remain authoritative — corrections happen via adjustments." });
   451→      setReopenOpen(false);
   452→    } catch (err) {
   453→      toast.error(errMessage(err));
   454→    } finally {
   455→      setActing(false);
   456→    }
   457→  }
   458→
   459→  if (isLoading) {
   460→    return (
   461→      <GlassCard className="space-y-4 p-4 sm:p-6">
   462→        <SectionHeading>Snapshot · {period.monthLabel}</SectionHeading>
   463→        <ListSkeleton rows={4} />
   464→      </GlassCard>
   465→    );
   466→  }
   467→
   468→  if (error || !data) {
   469→    return (
   470→      <GlassCard className="p-4 sm:p-6">
   471→        <SectionHeading>Snapshot · {period.monthLabel}</SectionHeading>
   472→        <div className="mt-3">
   473→          <ErrorState code={(error as ApiClientError | undefined)?.code} message={(error as ApiClientError | undefined)?.message} onRetry={() => void refetch()} />
   474→        </div>
   475→      </GlassCard>
   476→    );
   477→  }
   478→
   479→  return (
   480→    <GlassCard className="space-y-5 p-4 sm:p-6">
   481→      <SectionHeading
   482→        action={
   483→          reopenWindowOpen ? (
   484→            <GlassButton variant="destructive" size="sm" icon={<RotateCcw />} onClick={() => setReopenOpen(true)}>
   485→              Reopen…
   486→            </GlassButton>
   487→          ) : (
   488→            <span className="text-[11px] text-muted-foreground">Reopen window closed</span>
   489→          )
   490→        }
   491→      >
   492→        Snapshot · {period.monthLabel} · billed {period.billedAt ? fmtDate(period.billedAt) : "—"}
   493→      </SectionHeading>
   494→
   495→      {data.snapshot && (
   496→        <div className="grid grid-cols-2 gap-3 min-[420px]:grid-cols-4">
   497→          <KeyValue stacked label="Residents" value={<span className="kpi-num text-base font-semibold">{data.snapshot.residentCount}</span>} />
   498→          <KeyValue stacked label="Resident meals" value={<span className="kpi-num text-base font-semibold">{data.snapshot.residentMealCount}</span>} />
   499→          <KeyValue stacked label="Guest meals" value={<span className="kpi-num text-base font-semibold">{data.snapshot.guestMealCount}</span>} />
   500→          <KeyValue stacked label="Meal charge" value={<span className="kpi-num text-base font-semibold">{data.snapshot.mealChargeFormatted}</span>} />
   501→          <KeyValue stacked label="Expenses" value={<span className="kpi-num text-base font-semibold">{data.snapshot.eligibleExpensesFormatted}</span>} />

   639→function AdjustmentDialog({
   640→  bill,
   641→  tz,
   642→  onClose,
   643→  onSaved,
   644→}: {
   645→  bill: BillRow;
   646→  tz: string;
   647→  onClose: () => void;
   648→  onSaved: () => void;
   649→}) {
   650→  const [amount, setAmount] = useState("");
   651→  const [reason, setReason] = useState("");
   652→  const [saving, setSaving] = useState(false);
   653→  const [fields, setFields] = useState<Record<string, string>>({});
   654→
   655→  const amountValid = /^-?\d+(\.\d{1,2})?$/.test(amount.trim()) && Number(amount) !== 0;
   656→  const reasonValid = reason.trim().length >= 3;
   657→
   658→  async function save() {
   659→    setSaving(true);
   660→    setFields({});
   661→    try {
   662→      await postJson(`/api/v1/admin/bills/${bill.id}/adjustment`, {
   663→        amount: amount.trim(),
   664→        reason: reason.trim(),
   665→      });
   666→      toast.success("Bill adjusted", {
   667→        description: `${bill.billNumber} · ${Number(amount) < 0 ? "reduced" : "increased"} by ₹${Math.abs(Number(amount)).toFixed(2)}`,
   668→      });
   669→      onSaved();
   670→      onClose();
   671→    } catch (err) {
   672→      if (err instanceof ApiClientError && err.fields) setFields(err.fields);
   673→      toast.error(errMessage(err));
   674→    } finally {
   675→      setSaving(false);
   676→    }
   677→  }
   678→
   679→  return (
   680→    <Dialog open onOpenChange={(open) => !open && onClose()}>
   681→      <DialogContent className="glass-strong rounded-[28px] border-0 p-0 sm:max-w-md">
   682→        <div className="flex max-h-[82vh] flex-col">
   683→          <div className="px-5 pt-5 sm:px-6 sm:pt-6">
   684→            <DialogTitle className="text-left text-lg font-semibold tracking-tight">Adjust {bill.billNumber}</DialogTitle>
   685→            <DialogDescription className="mt-1.5 text-left text-[13px] leading-relaxed text-muted-foreground">
   686→              Closed-period corrections happen through adjustments only — the original bill stays intact and the change
   687→              is audited.
   688→            </DialogDescription>
   689→          </div>
   690→          <div className="min-h-0 flex-1 overflow-y-auto px-5 py-4 sm:px-6">
   691→            <div className="space-y-4">
   692→              <div className="glass-inset grid grid-cols-3 gap-2 rounded-md p-3.5">
   693→                <KeyValue stacked label="Subtotal" value={<Money minor={bill.subtotalMinor} />} />
   694→                <KeyValue stacked label="Paid" value={<Money minor={bill.paymentsMinor} />} />
   695→                <KeyValue stacked label="Due" value={<Money minor={bill.totalDueMinor} />} />
   696→              </div>
   697→              <MoneyField
   698→                label="Adjustment amount"
   699→                value={amount}
   700→                onChange={setAmount}
   701→                allowNegative
   702→                placeholder="-50.00 or 50.00"
   703→                error={fields.amount ?? (amount.trim() !== "" && !amountValid ? "Use a non-zero amount like -50.00 or 50.00." : undefined)}
   704→                hint="Negative reduces what the resident owes; positive increases it."
   705→              />
   706→              <TextAreaField
   707→                label="Reason"
   708→                value={reason}
   709→                onChange={setReason}
   710→                rows={2}
   711→                maxLength={500}
   712→                placeholder="Why is this correction needed? (required)"
   713→                error={fields.reason ?? (reason.trim() !== "" && !reasonValid ? "A short reason is required." : undefined)}
   714→              />
   715→              <p className="kpi-num text-[11px] text-muted-foreground/75">
   716→                Generated {fmtDateTime(bill.generatedAt, tz)}
   717→              </p>
   718→            </div>
   719→          </div>
   720→          <div className="safe-b flex items-center justify-end gap-2 border-t border-border/50 px-5 py-4 sm:px-6">
   721→            <GlassButton variant="ghost" onClick={onClose} disabled={saving}>
   722→              Cancel
   723→            </GlassButton>
   724→            <GlassButton variant="primary" loading={saving} disabled={!amountValid || !reasonValid} onClick={() => void save()}>
   725→              Apply adjustment
   726→            </GlassButton>
   727→          </div>
   728→        </div>
   729→      </DialogContent>
   730→    </Dialog>
   731→  );
   732→}
