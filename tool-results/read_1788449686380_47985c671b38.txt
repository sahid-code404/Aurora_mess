    70→interface MealsCacheValue {
    71→  data: MealInstanceDto[];
    72→  meta: MealsMeta;
    73→}
    74→
    75→function patchCacheRows(
    76→  value: MealsCacheValue | undefined,
    77→  instanceId: string,
    78→  patch: (m: MealInstanceDto) => MealInstanceDto
    79→): MealsCacheValue | undefined {
    80→  if (!value) return value;
    81→  return { ...value, data: value.data.map((m) => (m.id === instanceId ? patch(m) : m)) };
    82→}

    86→/** (year, monthIndex 0-11) → "YYYY-MM". */
    87→function monthKeyOf(year: number, month: number): string {
    88→  return `${year}-${String(month + 1).padStart(2, "0")}`;
    89→}
    90→
    91→function monthStartKey(year: number, month: number): string {
    92→  return `${monthKeyOf(year, month)}-01`;
    93→}
    94→
    95→function monthEndKey(year: number, month: number): string {
    96→  const lastDay = new Date(Date.UTC(year, month + 1, 0)).getUTCDate();
    97→  return `${monthKeyOf(year, month)}-${String(lastDay).padStart(2, "0")}`;
    98→}

   100→/** "September 2025". */
   101→function monthLongLabel(year: number, month: number): string {
   102→  return new Intl.DateTimeFormat("en-IN", { month: "long", year: "numeric", timeZone: "UTC" }).format(
   103→    new Date(Date.UTC(year, month, 1))
   104→  );
   105→}
   106→
   107→/** "Sep 2025" — compact KPI sub for non-current months. */
   108→function monthShortKey(year: number, month: number): string {
   109→  return new Intl.DateTimeFormat("en-IN", { month: "short", year: "numeric", timeZone: "UTC" }).format(
   110→    new Date(Date.UTC(year, month, 1))
   111→  );
   112→}

   114→/** Picker pill labels — relative for ±1 day, compact otherwise (BoardOps pattern). */
   115→function dayPillLabels(key: string, todayKey: string): { top: string; bottom: string } {
   116→  if (key === todayKey) return { top: "Today", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   117→  if (key === addDaysToKey(todayKey, -1)) return { top: "Yesterday", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   118→  if (key === addDaysToKey(todayKey, 1)) return { top: "Tomorrow", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   119→  return { top: formatDateInTz(key, "UTC"), bottom: weekdayShort(key) };
   120→}

   226→  async function handleToggle(meal: MealInstanceDto, next: "ON" | "OFF") {
   227→    const queries = queryClient.getQueriesData<MealsCacheValue>({
   228→      queryKey: ["api", "/api/v1/meals"],
   229→    });
   230→    const snapshots = queries.map(([key, value]) => ({ key, value }));
   231→
   232→    let current: MealInstanceDto | undefined;
   233→    for (const { value } of snapshots) {
   234→      current = value?.data.find((m) => m.id === meal.id);
   235→      if (current) break;
   236→    }
   237→    if (!current) return;
   238→
   239→    const expectedVersion = current.myState.version;
   240→
   241→    // Optimistically flip every cached copy of this row.
   242→    for (const { key } of snapshots) {
   243→      queryClient.setQueryData<MealsCacheValue>(key, (old) =>
   244→        patchCacheRows(old, meal.id, (m) => ({
   245→          ...m,
   246→          myState: { ...m.myState, effectiveState: next } as MealMyState,
   247→        }))
   248→      );
   249→    }
   250→
   251→    try {
   252→      const res = await apiJson<ToggleResponse>(`/api/v1/meals/${meal.id}/toggle`, "POST", {
   253→        state: next,
   254→        expectedVersion,
   255→      });
   256→      // Authoritative response wins.
   257→      for (const { key } of snapshots) {
   258→        queryClient.setQueryData<MealsCacheValue>(key, (old) =>
   259→          patchCacheRows(old, meal.id, (m) => ({
   260→            ...m,
   261→            myState: {
   262→              ...m.myState,
   263→              effectiveState: res.state,
   264→              effectiveReason: res.effectiveReason,
   265→              locked: res.locked,
   266→              version: res.version,
   267→            } as MealMyState,
   268→          }))
   269→        );
   270→      }
   271→      invalidate([RESIDENT_KEYS.dashboard]);
   272→    } catch (err) {
   273→      // Rollback — the toggle springs back (spec §114).
   274→      for (const { key, value } of snapshots) {
   275→        queryClient.setQueryData<MealsCacheValue>(key, value);
   276→      }
   277→      if (err instanceof ApiClientError) {
   278→        if (err.code === "MEAL_CUTOFF_PASSED") {
   279→          flashFor(meal.id, {
   280→            tone: "warning",
   281→            text: `${meal.name} is already locked. Admin can still override it.`,
   282→          });
   283→        } else if (err.code === "RESOURCE_CHANGED") {
   284→          flashFor(meal.id, { tone: "info", text: "This meal was just changed. Refreshing…" });
   285→          void queryClient.invalidateQueries({ queryKey: ["api", "/api/v1/meals"] });
   286→        } else if (err.code === "MEAL_NOT_AVAILABLE") {
   287→          flashFor(meal.id, { tone: "warning", text: "This meal isn't available to change right now." });
   288→        } else {
   289→          flashFor(meal.id, { tone: "danger", text: err.message });
   290→        }
   291→      } else {
   292→        flashFor(meal.id, { tone: "danger", text: friendlyError(err) });
   293→      }
   294→    }
   295→  }

   359→  /**
   360→   * Adjust the day's guest count with the ± stepper — self-service, under
   361→   * cutoff, no admin permission (spec: guest meals behave like normal
   362→   * meals). Optimistic patch of every cached guest list copy with rollback
   363→   * and a plain-language flash, exactly like the meal toggles (spec §114).
   364→   */
   365→  async function handleGuestStep(dateKey: string, delta: 1 | -1) {
   366→    const dayGuests = guestsByDate.get(dateKey) ?? [];
   367→    const queries = queryClient.getQueriesData<{ data: GuestMealDto[]; meta: unknown }>({
   368→      queryKey: ["api", "/api/v1/guest-meals"],
   369→    });
   370→    const snapshots = queries.map(([key, value]) => ({ key, value }));
   371→
   372→    // Optimistic: apply the same target pick locally.
   373→    const target = pickGuestStepTarget(dayGuests, delta, serverNow);
   374→    if (target) {
   375→      const nextQty = delta === 1 ? target.quantity + 1 : target.quantity - 1;
   376→      for (const { key } of snapshots) {
   377→        queryClient.setQueryData<{ data: GuestMealDto[]; meta: unknown }>(key, (old) =>
   378→          old
   379→            ? {
   380→                ...old,
   381→                data: old.data.map((g) =>
   382→                  g.id !== target.id
   383→                    ? g
   384→                    : nextQty <= 0
   385→                      ? { ...g, status: "CANCELLED" }
   386→                      : { ...g, quantity: nextQty, totalPriceMinor: g.unitPriceMinor * nextQty }
   387→                ),
   388→              }
   389→            : old
   390→        );
   391→      }
   392→    }
   393→
   394→    const result = await stepGuestMeals(dayGuests, delta, serverNow);
   395→    if (result.kind === "ok") {
   396→      invalidate([RESIDENT_KEYS.guestMeals, RESIDENT_KEYS.dashboard, RESIDENT_KEYS.billing, RESIDENT_KEYS.notifications]);
   397→    } else if (result.kind === "dialog") {
   398→      // No request can absorb the +1 — the meal choice dialog takes over.
   399→      for (const { key, value } of snapshots) queryClient.setQueryData(key, value);
   400→      openGuestDialog(dateKey);
   401→    } else {
   402→      // Rollback — the counter springs back.
   403→      for (const { key, value } of snapshots) queryClient.setQueryData(key, value);
   404→      if (result.code === "RESOURCE_CHANGED") {
   405→        flashFor(`guest-${dateKey}`, { tone: "info", text: "This guest meal was just changed. Refreshing…" });
   406→        void queryClient.invalidateQueries({ queryKey: ["api", "/api/v1/guest-meals"] });
   407→      } else if (result.code === "MEAL_CUTOFF_PASSED") {
   408→        flashFor(`guest-${dateKey}`, {
   409→          tone: "warning",
   410→          text: "Guest meals for this day are locked — the cutoff already passed.",
   411→        });
   412→      } else {
   413→        flashFor(`guest-${dateKey}`, { tone: "danger", text: result.message });
   414→      }
   415→    }
   416→  }
   417→
   418→  const stats = useMemo(() => {
   419→    let on = 0;
   420→    let off = 0;
   421→    let locked = 0;
   422→    for (const m of meals) {
   423→      if (m.myState.effectiveState === "ON") on++;
   424→      else if (m.myState.effectiveState === "OFF") off++;
   425→      if (m.myState.locked) locked++;
   426→    }

   446→    }

   448→  }, [dayMeals]);

   461→  }, [view, mealsQuery.isPending, dayMap, todayKey]);

   472→    setExpandedDay(null);
   473→  }
