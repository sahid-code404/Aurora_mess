    76→    76→  value: MealsCacheValue | undefined,

    79→    79→): MealsCacheValue | undefined {

    87→    87→function monthKeyOf(year: number, month: number): string {
    88→    88→  return `${year}-${String(month + 1).padStart(2, "0")}`;
    89→    89→}

    91→    91→function monthStartKey(year: number, month: number): string {

    98→    98→}

   114→   114→/** Picker pill labels — relative for ±1 day, compact otherwise (BoardOps pattern). */
   115→   115→function dayPillLabels(key: string, todayKey: string): { top: string; bottom: string } {
   116→   116→  if (key === todayKey) return { top: "Today", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   117→   117→  if (key === addDaysToKey(todayKey, -1)) return { top: "Yesterday", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   118→   118→  if (key === addDaysToKey(todayKey, 1)) return { top: "Tomorrow", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   119→   119→  return { top: formatDateInTz(key, "UTC"), bottom: weekdayShort(key) };

   225→   225→
   226→   226→  async function handleToggle(meal: MealInstanceDto, next: "ON" | "OFF") {
   227→   227→    const queries = queryClient.getQueriesData<MealsCacheValue>({
   228→   228→      queryKey: ["api", "/api/v1/meals"],
   229→   229→    });
   230→   230→    const snapshots = queries.map(([key, value]) => ({ key, value }));
   231→   231→
   232→   232→    let current: MealInstanceDto | undefined;
   233→   233→    for (const { value } of snapshots) {
   234→   234→      current = value?.data.find((m) => m.id === meal.id);
   235→   235→      if (current) break;
   236→   236→    }
   237→   237→    if (!current) return;
   238→   238→
   239→   239→    const expectedVersion = current.myState.version;
   240→   240→
   241→   241→    // Optimistically flip every cached copy of this row.
   242→   242→    for (const { key } of snapshots) {
   243→   243→      queryClient.setQueryData<MealsCacheValue>(key, (old) =>
   244→   244→        patchCacheRows(old, meal.id, (m) => ({
   245→   245→          ...m,
   246→   246→          myState: { ...m.myState, effectiveState: next } as MealMyState,
   247→   247→        }))
   248→   248→      );
   249→   249→    }
   250→   250→
   251→   251→    try {
   252→   252→      const res = await apiJson<ToggleResponse>(`/api/v1/meals/${meal.id}/toggle`, "POST", {
   253→   253→        state: next,
   254→   254→        expectedVersion,
   255→   255→      });
   256→   256→      // Authoritative response wins.
   257→   257→      for (const { key } of snapshots) {
   258→   258→        queryClient.setQueryData<MealsCacheValue>(key, (old) =>
   259→   259→          patchCacheRows(old, meal.id, (m) => ({
   260→   260→            ...m,
   261→   261→            myState: {
   262→   262→              ...m.myState,
   263→   263→              effectiveState: res.state,
   264→   264→              effectiveReason: res.effectiveReason,
   265→   265→              locked: res.locked,
   266→   266→              version: res.version,
   267→   267→            } as MealMyState,
   268→   268→          }))
   269→   269→        );
   270→   270→      }
   271→   271→      invalidate([RESIDENT_KEYS.dashboard]);
   272→   272→    } catch (err) {
   273→   273→      // Rollback — the toggle springs back (spec §114).
   274→   274→      for (const { key, value } of snapshots) {
   275→   275→        queryClient.setQueryData<MealsCacheValue>(key, value);
   276→   276→      }
   277→   277→      if (err instanceof ApiClientError) {
   278→   278→        if (err.code === "MEAL_CUTOFF_PASSED") {
   279→   279→          flashFor(meal.id, {
   280→   280→            tone: "warning",
   281→   281→            text: `${meal.name} is already locked. Admin can still override it.`,
   282→   282→          });
   283→   283→        } else if (err.code === "RESOURCE_CHANGED") {
   284→   284→          flashFor(meal.id, { tone: "info", text: "This meal was just changed. Refreshing…" });
   285→   285→          void queryClient.invalidateQueries({ queryKey: ["api", "/api/v1/meals"] });
   286→   286→        } else if (err.code === "MEAL_NOT_AVAILABLE") {
   287→   287→          flashFor(meal.id, { tone: "warning", text: "This meal isn't available to change right now." });
   288→   288→        } else {
   289→   289→          flashFor(meal.id, { tone: "danger", text: err.message });
   290→   290→        }
   291→   291→      } else {
   292→   292→        flashFor(meal.id, { tone: "danger", text: friendlyError(err) });
   293→   293→      }
   294→   294→    }
   295→   295→  }

   359→   359→  /**
   360→   360→   * Adjust the day's guest count with the ± stepper — self-service, under
   361→   361→   * cutoff, no admin permission (spec: guest meals behave like normal
   362→   362→   * meals). Optimistic patch of every cached guest list copy with rollback
   363→   363→   * and a plain-language flash, exactly like the meal toggles (spec §114).
   364→   364→   */
   365→   365→  async function handleGuestStep(dateKey: string, delta: 1 | -1) {
   366→   366→    const dayGuests = guestsByDate.get(dateKey) ?? [];
   367→   367→    const queries = queryClient.getQueriesData<{ data: GuestMealDto[]; meta: unknown }>({
   368→   368→      queryKey: ["api", "/api/v1/guest-meals"],
   369→   369→    });
   370→   370→    const snapshots = queries.map(([key, value]) => ({ key, value }));
   371→   371→
   372→   372→    // Optimistic: apply the same target pick locally.
   373→   373→    const target = pickGuestStepTarget(dayGuests, delta, serverNow);
   374→   374→    if (target) {
   375→   375→      const nextQty = delta === 1 ? target.quantity + 1 : target.quantity - 1;
   376→   376→      for (const { key } of snapshots) {
   377→   377→        queryClient.setQueryData<{ data: GuestMealDto[]; meta: unknown }>(key, (old) =>
   378→   378→          old
   379→   379→            ? {
   380→   380→                ...old,
   381→   381→                data: old.data.map((g) =>
   382→   382→                  g.id !== target.id
   383→   383→                    ? g
   384→   384→                    : nextQty <= 0
   385→   385→                      ? { ...g, status: "CANCELLED" }
   386→   386→                      : { ...g, quantity: nextQty, totalPriceMinor: g.unitPriceMinor * nextQty }
   387→   387→                ),
   388→   388→              }
   389→   389→            : old
   390→   390→        );
   391→   391→      }
   392→   392→    }
   393→   393→
   394→   394→    const result = await stepGuestMeals(dayGuests, delta, serverNow);
   395→   395→    if (result.kind === "ok") {
   396→   396→      invalidate([RESIDENT_KEYS.guestMeals, RESIDENT_KEYS.dashboard, RESIDENT_KEYS.billing, RESIDENT_KEYS.notifications]);
   397→   397→    } else if (result.kind === "dialog") {
   398→   398→      // No request can absorb the +1 — the meal choice dialog takes over.
   399→   399→      for (const { key, value } of snapshots) queryClient.setQueryData(key, value);
   400→   400→      openGuestDialog(dateKey);
   401→   401→    } else {
   402→   402→      // Rollback — the counter springs back.
   403→   403→      for (const { key, value } of snapshots) queryClient.setQueryData(key, value);
   404→   404→      if (result.code === "RESOURCE_CHANGED") {
   405→   405→        flashFor(`guest-${dateKey}`, { tone: "info", text: "This guest meal was just changed. Refreshing…" });
   406→   406→        void queryClient.invalidateQueries({ queryKey: ["api", "/api/v1/guest-meals"] });
   407→   407→      } else if (result.code === "MEAL_CUTOFF_PASSED") {
   408→   408→        flashFor(`guest-${dateKey}`, {
   409→   409→          tone: "warning",
   410→   410→          text: "Guest meals for this day are locked — the cutoff already passed.",
   411→   411→        });
   412→   412→      } else {
   413→   413→        flashFor(`guest-${dateKey}`, { tone: "danger", text: result.message });
   414→   414→      }
   415→   415→    }
   416→   416→  }
   417→   417→
   418→   418→  const stats = useMemo(() => {
   419→   419→    let on = 0;
   420→   420→    let off = 0;
   421→   421→    let locked = 0;
   422→   422→    for (const m of meals) {
   423→   423→      if (m.myState.effectiveState === "ON") on++;
   424→   424→      else if (m.myState.effectiveState === "OFF") off++;
   425→   425→      if (m.myState.locked) locked++;
   426→   426→    }
   427→   427→    return { on, off, locked };
   428→   428→  }, [meals]);
   429→   429→
   430→   430→  const dayMeals = useMemo(
   431→   431→    () =>
