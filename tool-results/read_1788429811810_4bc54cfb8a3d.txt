    98→/** (year, monthIndex 0-11) → "YYYY-MM". */
    99→function monthKeyOf(year: number, month: number): string {
   100→  return `${year}-${String(month + 1).padStart(2, "0")}`;
   101→}
   102→
   103→function monthStartKey(year: number, month: number): string {
   104→  return `${monthKeyOf(year, month)}-01`;
   105→}
   106→
   107→function monthEndKey(year: number, month: number): string {
   108→  const lastDay = new Date(Date.UTC(year, month + 1, 0)).getUTCDate();
   109→  return `${monthKeyOf(year, month)}-${String(lastDay).padStart(2, "0")}`;
   110→}
   111→
   112→/** "September 2025". */
   113→function monthLongLabel(year: number, month: number): string {
   114→  return new Intl.DateTimeFormat("en-IN", { month: "long", year: "numeric", timeZone: "UTC" }).format(
   115→    new Date(Date.UTC(year, month, 1))
   116→  );
   117→}
   118→
   119→/** "Sep 2025" — compact KPI sub for non-current months. */
   120→function monthShortKey(year: number, month: number): string {
   121→  return new Intl.DateTimeFormat("en-IN", { month: "short", year: "numeric", timeZone: "UTC" }).format(
   122→    new Date(Date.UTC(year, month, 1))
   123→  );
   124→}
   125→
   126→function parseKey(key: string): Date {
   127→  const [y, m, d] = key.split("-").map(Number);
   128→  return new Date(Date.UTC(y, (m ?? 1) - 1, d ?? 1));
   129→}
   130→
   131→function weekdayShort(key: string): string {
   132→  return new Intl.DateTimeFormat("en-IN", { weekday: "short", timeZone: "UTC" }).format(parseKey(key));
   133→}
   134→
   135→function dayNum(key: string): string {
   136→  return String(parseKey(key).getUTCDate());
   137→}
   138→
   139→/** "Monday, 3 September". */
   140→function longDayLabel(key: string): string {
   141→  return new Intl.DateTimeFormat("en-IN", { weekday: "long", day: "numeric", month: "long", timeZone: "UTC" }).format(
   142→    parseKey(key)
   143→  );
   144→}
   145→
   146→/** Picker pill labels — relative for ±1 day, compact otherwise (BoardOps pattern). */
   147→function dayPillLabels(key: string, todayKey: string): { top: string; bottom: string } {
   148→  if (key === todayKey) return { top: "Today", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   149→  if (key === addDaysToKey(todayKey, -1)) return { top: "Yesterday", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   150→  if (key === addDaysToKey(todayKey, 1)) return { top: "Tomorrow", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   151→  return { top: formatDateInTz(key, "UTC"), bottom: weekdayShort(key) };
   152→}

   158→function MiniChip({
   159→  tone = "neutral",
   160→  icon: Icon,
   161→  children,
   162→}: {
   163→  tone?: "success" | "warning" | "danger" | "primary" | "neutral";
   164→  icon?: LucideIcon;
   165→  children: React.ReactNode;
   166→}) {
   167→  const tones: Record<string, string> = {
   168→    success: "bg-success/15 text-success",
   169→    warning: "bg-warning/15 text-warning",
   170→    danger: "bg-danger/15 text-danger",
   171→    primary: "bg-primary/15 text-primary",
   172→    neutral: "bg-muted/80 text-muted-foreground",
   173→  };
   174→  return (
   175→    <span
   176→      className={cn(
   177→        "inline-flex items-center gap-0.5 whitespace-nowrap rounded-pill px-1.5 py-0.5 text-[10px] font-medium",
   178→        tones[tone]
   179→      )}
   180→    >
   181→      {Icon && <Icon className="size-2.5" aria-hidden />}
   182→      {children}
   183→    </span>
   184→  );
   185→}

   222→function AgendaDayRow({
   223→  dateKey,
   224→  todayKey,
   225→  entries,
   226→  tz,
   227→  now,
   228→  expanded,
   229→  guestsByInstance,
   230→  flash,
   231→  onToggleExpand,
   232→  onToggleMeal,
   233→}: {
   234→  dateKey: string;
   235→  todayKey: string;
   236→  entries: MealInstanceDto[];
   237→  tz: string;
   238→  now: number;
   239→  expanded: boolean;
   240→  guestsByInstance: Map<string, number>;
   241→  flash?: Record<string, Flash>;
   242→  onToggleExpand: () => void;
   243→  onToggleMeal: (meal: MealInstanceDto, next: "ON" | "OFF") => void;
   244→}) {
   245→  const isToday = dateKey === todayKey;
   246→  const onCount = entries.filter((m) => m.myState.effectiveState === "ON").length;
   247→  const offCount = entries.filter((m) => m.myState.effectiveState === "OFF").length;
   248→  const leaveCount = entries.filter((m) => m.myState.effectiveState === "ON_LEAVE").length;
   249→  const lockedCount = entries.filter((m) => m.myState.locked).length;
   250→  const adminCount = entries.filter((m) => m.myState.overridden).length;
   251→
   252→  return (
   253→    <GlassCard className="overflow-hidden">
   254→      <button
   255→        type="button"
   256→        onClick={onToggleExpand}
   257→        aria-expanded={expanded}
   258→        className="flex w-full cursor-pointer items-center gap-4 p-4 text-left transition-colors hover:bg-foreground/[0.04] dark:hover:bg-white/[0.04]"
   259→      >
   260→        {/* Circular gradient date disc — the BoardOps day marker (today pops). */}
   261→        <div
   262→          className={cn(
   263→            "flex size-14 shrink-0 flex-col items-center justify-center rounded-full",
   264→            isToday
   265→              ? "bg-gradient-to-br from-primary/75 to-primary text-primary-foreground shadow-[inset_0_1px_0_rgba(255,255,255,0.35),0_8px_20px_-8px_color-mix(in_oklab,var(--primary)_50%,transparent)]"
   266→              : "glass-inset text-muted-foreground"
   267→          )}
   268→        >
   269→          <span className="text-[10px] font-bold uppercase leading-none tracking-wide">
   270→            {weekdayShort(dateKey)}
   271→          </span>
   272→          <span
   273→            className={cn(
   274→              "mt-1 text-xl font-bold leading-none kpi-num",
   275→              isToday ? "text-primary-foreground" : "text-foreground"
   276→            )}
   277→          >
   278→            {dayNum(dateKey)}
   279→          </span>
   280→        </div>
   281→        <div className="min-w-0 flex-1">
   282→          <p className={cn("truncate text-base font-semibold", isToday && "text-primary")}>
   283→            {isToday ? "Today" : longDayLabel(dateKey)}
   284→          </p>
   285→          <div className="mt-1.5 flex flex-wrap items-center gap-1.5">
   286→            {onCount > 0 && (
   287→              <MiniChip tone="success" icon={Check}>
   288→                {onCount} ON
   289→              </MiniChip>
   290→            )}
   291→            {offCount > 0 && (
   292→              <MiniChip tone="warning" icon={X}>
   293→                {offCount} OFF
   294→              </MiniChip>
   295→            )}
   296→            {leaveCount > 0 && <MiniChip tone="neutral" icon={Plane}>{leaveCount} on leave</MiniChip>}
   297→            {lockedCount > 0 && (
   298→              <MiniChip tone="danger" icon={Lock}>
   299→                {lockedCount}
   300→              </MiniChip>
   301→            )}
   302→            {adminCount > 0 && (
   303→              <MiniChip tone="primary" icon={ShieldCheck}>
   304→                {adminCount} admin
   305→              </MiniChip>
   306→            )}
   307→          </div>
   308→        </div>
   309→        <motion.span
   310→          animate={{ rotate: expanded ? 180 : 0 }}
   311→          transition={{ duration: 0.2 }}
   312→          className="flex shrink-0 items-center text-muted-foreground"
   313→        >
   314→          <ChevronDown className="size-4" aria-hidden />
   315→        </motion.span>
   316→      </button>
   317→
   318→      <AnimatePresence initial={false}>
   319→        {expanded && (
   320→          <motion.div
   321→            initial={{ height: 0, opacity: 0 }}
   322→            animate={{ height: "auto", opacity: 1 }}
   323→            exit={{ height: 0, opacity: 0 }}
   324→            transition={{ duration: 0.25, ease: [0.22, 1, 0.36, 1] }}
   325→            className="overflow-hidden"
   326→          >
   327→            <div className="space-y-2.5 px-4 pb-4">
   328→              {entries.map((meal) => (
   329→                <AgendaMealRow
   330→                  key={meal.id}
   331→                  meal={meal}
   332→                  tz={tz}
   333→                  now={now}
   334→                  guests={guestsByInstance.get(meal.id) ?? 0}
   335→                  flash={flash?.[meal.id]}
   336→                  onToggle={onToggleMeal}
   337→                />
   338→              ))}
   339→            </div>
   340→          </motion.div>
   341→        )}
   342→      </AnimatePresence>
   343→    </GlassCard>
   344→  );
   345→}

   347→/* ---------------------------- agenda meal row ------------------------------ */
   348→
   349→function AgendaMealRow({
   350→  meal,
   351→  tz,
   352→  now,
   353→  guests,
   354→  flash,
   355→  onToggle,
   356→}: {
   357→  meal: MealInstanceDto;
   358→  tz: string;
   359→  now: number;
   360→  guests: number;
   361→  flash?: Flash;
   362→  onToggle: (meal: MealInstanceDto, next: "ON" | "OFF") => void;
   363→}) {
   364→  const state = meal.myState.effectiveState;
   365→  const overridden = meal.myState.overridden;
   366→  const locked = meal.myState.locked || new Date(meal.cutoffAt).getTime() <= now;
   367→  const canToggle = !locked && !overridden && (state === "ON" || state === "OFF");
   368→  const cutoffLeft = new Date(meal.cutoffAt).getTime() - now;
   369→
   370→  return (
   371→    <div className="glass-inset rounded-2xl p-3">
   372→      <div className="flex items-center gap-3.5">
   373→        <MealOrb icon={<MealIcon name={meal.icon} />} colorToken={meal.colorToken} />
   374→        <div className="min-w-0 flex-1">
   375→          <p className="truncate text-[15px] font-semibold leading-tight">{meal.name}</p>
   376→          <div className="mt-1 flex flex-wrap items-center gap-x-2 gap-y-1 text-xs text-muted-foreground">
   377→            <span className="kpi-num">{formatWindowInTz(meal.serviceWindow.startAt, meal.serviceWindow.endAt, tz)}</span>
   378→            {canToggle && cutoffLeft > 0 && (
   379→              <>
   380→                <span aria-hidden>·</span>
   381→                <span className="kpi-num">{countdownLabel(cutoffLeft)}</span>
   382→              </>
   383→            )}
   384→            {locked && <Lock className="size-3 text-danger" aria-hidden />}
   385→            {overridden && (
   386→              <MiniChip tone="primary" icon={ShieldCheck}>
   387→                Admin
   388→              </MiniChip>
   389→            )}
   390→            {state === "ON_LEAVE" && <MiniChip tone="neutral" icon={Plane}>On leave</MiniChip>}
   391→            {state === "NOT_AVAILABLE" && meal.myState.effectiveReason in NOT_AVAILABLE_REASONS && (
   392→              <span className="font-medium">{NOT_AVAILABLE_REASONS[meal.myState.effectiveReason]}</span>
   393→            )}
   394→            {guests > 0 && (
   395→              <span className="kpi-num inline-flex items-center gap-1 font-medium text-primary">
   396→                <Users className="size-3" aria-hidden />
   397→                {guests}
   398→              </span>
   399→            )}
   400→          </div>
   401→        </div>
   402→        <div className="flex shrink-0 items-center gap-1.5">
   403→          {canToggle ? (
   404→            <GlassToggle
   405→              checked={state === "ON"}
   406→              onChange={(next) => onToggle(meal, next ? "ON" : "OFF")}
   407→              label={`${meal.name} on ${meal.serviceDate}: ${stateLabel(state)}`}
   408→            />
   409→          ) : (
   410→            <>
   411→              {overridden && <StatusBadge status="ADMIN_OVERRIDE" />}
   412→              <StatusBadge status={state} label={stateLabel(state)} />
   413→              {locked && <StatusBadge status="LOCKED" />}
   414→            </>
   415→          )}
   416→        </div>
   417→      </div>
   418→
   419→      <AnimatePresence>
   420→        {flash && (
   421→          <motion.div
   422→            initial={{ opacity: 0, height: 0 }}
   423→            animate={{ opacity: 1, height: "auto" }}
   424→            exit={{ opacity: 0, height: 0 }}
   425→            transition={{ duration: 0.22 }}
   426→            className="overflow-hidden"
   427→          >
   428→            <div className="px-3 pb-3">
   429→              <FormNotice tone={flash.tone === "info" ? "info" : flash.tone}>{flash.text}</FormNotice>
   430→            </div>
   431→          </motion.div>
   432→        )}
   433→      </AnimatePresence>
   434→    </div>
   435→  );
   436→}

   606→  async function handleToggle(meal: MealInstanceDto, next: "ON" | "OFF") {
   607→    const queries = queryClient.getQueriesData<MealsCacheValue>({
   608→      queryKey: ["api", "/api/v1/meals"],
   609→    });

   612→    let current: MealInstanceDto | undefined;

   614→      current = value?.data.find((m) => m.id === meal.id);
   615→      if (current) break;
   616→    }
   617→    if (!current) return;
   618→
   619→    const expectedVersion = current.myState.version;
   620→
   621→    // Optimistically flip every cached copy of this row.
   622→    for (const { key } of snapshots) {
   623→      queryClient.setQueryData<MealsCacheValue>(key, (old) =>
   624→        patchCacheRows(old, meal.id, (m) => ({
   625→          ...m,
   626→          myState: { ...m.myState, effectiveState: next } as MealMyState,
   627→        }))
   628→      );
   629→    }
   630→
   631→    try {
   632→      const res = await apiJson<ToggleResponse>(`/api/v1/meals/${meal.id}/toggle`, "POST", {
   633→        state: next,
   634→        expectedVersion,
   635→      });
   636→      // Authoritative response wins.
   637→      for (const { key } of snapshots) {
   638→        queryClient.setQueryData<MealsCacheValue>(key, (old) =>
   639→          patchCacheRows(old, meal.id, (m) => ({
   640→            ...m,
   641→            myState: {
   642→              ...m.myState,
   643→              effectiveState: res.state,
   644→              effectiveReason: res.effectiveReason,
   645→              locked: res.locked,
   646→              version: res.version,
   647→            } as MealMyState,
   648→          }))
   649→        );
   650→      }
   651→      invalidate([RESIDENT_KEYS.dashboard]);
   652→    } catch (err) {
   653→      // Rollback — the toggle springs back (spec §114).
   654→      for (const { key, value } of snapshots) {
   655→        queryClient.setQueryData<MealsCacheValue>(key, value);
   656→      }

   658→        if (err.code === "MEAL_CUTOFF_PASSED") {
   659→          flashFor(meal.id, {
   660→            tone: "warning",
   661→            text: `${meal.name} is already locked. Admin can still override it.`,
   662→          });
   663→        } else if (err.code === "RESOURCE_CHANGED") {

   669→          flashFor(meal.id, { tone: "danger", text: err.message });
