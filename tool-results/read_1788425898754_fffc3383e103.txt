   152→async function unwrap<T>(promise: Promise<unknown>): Promise<T> {
   153→  const res = await promise;
   154→  if (res && typeof res === "object" && "success" in res && "data" in (res as Record<string, unknown>)) {
   155→    return (res as unknown as { data: T }).data;
   156→  }
   157→  return res as T;
   158→}

   160→function initials(name: string) {
   161→  return name
   162→    .split(" ")
   163→    .filter(Boolean)
   164→    .slice(0, 2)
   165→    .map((p) => p[0]?.toUpperCase() ?? "")
   166→    .join("");
   167→}

   985→function actionResultStatus(action: Action): string {
   986→  switch (action) {
   987→    case "APPROVE":
   988→    case "ACTIVATE":
   989→    case "RESTORE":
   990→      return "Active";
   991→    case "SUSPEND":
   992→      return "Suspended";
   993→    case "DEACTIVATE":
   994→      return "Inactive";
   995→    case "ARCHIVE":
   996→    case "REJECT":
   997→      return "Archived";
   998→    default:
   999→      return "";
  1000→  }
