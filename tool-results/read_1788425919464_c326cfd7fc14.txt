   106→function gradientFor(name: string) {
   107→  const idx = name.split("").reduce((acc, c) => acc + c.charCodeAt(0), 0) % AVATAR_GRADIENTS.length;
   108→  return AVATAR_GRADIENTS[idx];

   111→function initials(name: string) {
   112→  return name
   113→    .split(" ")
   114→    .filter(Boolean)
   115→    .slice(0, 2)
   116→    .map((p) => p[0]?.toUpperCase() ?? "")
   117→    .join("");

   120→async function unwrap<T>(promise: Promise<unknown>): Promise<T> {
   121→  const res = await promise;
   122→  if (res && typeof res === "object" && "success" in res && "data" in (res as Record<string, unknown>)) {
   123→    return (res as unknown as { data: T }).data;
   124→  }
   125→  return res as T;
   126→}

   511→function AvatarUpload({
   512→  avatarUrl,
   513→  name,
   514→  onUpload,
   515→  loading,
   516→}: {
   517→  avatarUrl?: string;
   518→  name: string;
   519→  onUpload: (file: File) => void;
   520→  loading: boolean;
   521→}) {
   522→  const inputRef = useRef<HTMLInputElement>(null);
   523→
   524→  const handleFile = (file: File) => {
   525→    if (!file.type.startsWith("image/")) {
   526→      toast.error("Please select an image file");
   527→      return;
   528→    }
   529→    if (file.size > 4 * 1024 * 1024) {
   530→      toast.error("Image must be under 4 MB");
   531→      return;
   532→    }
   533→    onUpload(file);
   534→  };
   535→
   536→  return (
   537→    <motion.div
   538→      initial={{ scale: 0.85, opacity: 0 }}
   539→      animate={{ scale: 1, opacity: 1 }}
   540→      transition={{ type: "spring", stiffness: 220, damping: 18 }}
   541→      className="relative shrink-0 mx-auto"
   542→    >
   543→      <div className="absolute -inset-1 rounded-3xl bg-gradient-to-br from-primary/40 to-success/40 blur-md" />
   544→      <Avatar className="relative h-24 w-24 rounded-3xl">
   545→        {avatarUrl && <AvatarImage src={avatarUrl} alt={name} />}
   546→        <AvatarFallback
   547→          className={cn(
   548→            "rounded-3xl bg-gradient-to-br text-white font-bold text-2xl",
   549→            gradientFor(name)
   550→          )}
   551→        >
   552→          {initials(name) || "U"}
   553→        </AvatarFallback>
   554→      </Avatar>
   555→      <input
   556→        ref={inputRef}
   557→        type="file"
   558→        accept="image/jpeg,image/png,image/webp,image/gif"
   559→        className="hidden"
   560→        onChange={(e) => {
   561→          const file = e.target.files?.[0];
   562→          if (file) handleFile(file);
   563→          e.target.value = "";
   564→        }}
   565→      />
   566→      <button
   567→        onClick={() => inputRef.current?.click()}
   568→        disabled={loading}
   569→        className="absolute -bottom-1 -right-1 h-9 w-9 rounded-full glass-strong grid place-items-center ring-2 ring-background hover:scale-110 transition-transform disabled:opacity-50"

   892→function ChangePasswordDialog({
   893→  open,
   894→  onOpenChange,
   895→  onSuccess,
   896→}: {
   897→  open: boolean;
   898→  onOpenChange: (v: boolean) => void;
   899→  onSuccess: () => void;
   900→}) {
   901→  const isMobile = useIsMobile();
   902→  const [form, setForm] = useState({ currentPassword: "", newPassword: "", confirmPassword: "" });
   903→  const [showCurrent, setShowCurrent] = useState(false);
   904→  const [showNew, setShowNew] = useState(false);
   905→  const [loading, setLoading] = useState(false);
   906→
   907→  const passwordStrength = (() => {
   908→    const p = form.newPassword;
   909→    if (!p) return { score: 0, label: "", color: "" };
   910→    let score = 0;
   911→    if (p.length >= 8) score++;
   912→    if (/[A-Z]/.test(p)) score++;
   913→    if (/[a-z]/.test(p)) score++;
   914→    if (/[0-9]/.test(p)) score++;
   915→    if (/[^A-Za-z0-9]/.test(p)) score++;
   916→    const labels = ["", "Weak", "Fair", "Good", "Strong", "Excellent"];
   917→    const colors = ["", "bg-destructive", "bg-warning", "bg-warning", "bg-success", "bg-success"];
   918→    return { score, label: labels[score], color: colors[score] };
   919→  })();
   920→
   921→  const submit = async () => {
   922→    if (form.newPassword !== form.confirmPassword) {
   923→      toast.error("Passwords don't match");
   924→      return;
   925→    }
   926→    if (form.newPassword.length < 8) {
   927→      toast.error("Password must be at least 8 characters");
   928→      return;
   929→    }
   930→    setLoading(true);
   931→    try {
   932→      await api.post("/auth/change-password", {
   933→        currentPassword: form.currentPassword,
   934→        newPassword: form.newPassword,
   935→      });
   936→      toast.success("Password changed. Other sessions have been signed out.");
   937→      setForm({ currentPassword: "", newPassword: "", confirmPassword: "" });
   938→      onSuccess();
   939→    } catch (e: unknown) {
   940→      toast.error((e as Error).message || "Failed to change password");

   942→      setLoading(false);
   943→    }

   947→    <div className="space-y-4">
   948→      <GlassInput
   949→        label="Current Password"
   950→        type={showCurrent ? "text" : "password"}
   951→        value={form.currentPassword}
   952→        onChange={(e) => setForm({ ...form, currentPassword: e.target.value })}
   953→        icon={<Lock />}
   954→        trailing={
   955→          <button type="button" onClick={() => setShowCurrent(!showCurrent)} className="text-muted-foreground hover:text-foreground">
   956→            {showCurrent ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   957→          </button>
   958→        }
   959→      />
   960→      <GlassInput
   961→        label="New Password"
   962→        type={showNew ? "text" : "password"}

   964→        onChange={(e) => setForm({ ...form, newPassword: e.target.value })}
   965→        icon={<KeyRound />}
   966→        trailing={
   967→          <button type="button" onClick={() => setShowNew(!showNew)} className="text-muted-foreground hover:text-foreground">
   968→            {showNew ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   969→          </button>
   970→        }
   971→        hint="Min 8 chars with uppercase, lowercase, and a number"
   972→      />
   973→      {form.newPassword && (
   974→        <div className="flex items-center gap-2">
   975→          <div className="flex-1 h-1.5 rounded-full bg-muted overflow-hidden">

  1050→function TwoFactorDialog({
  1051→  open,
  1052→  onOpenChange,
  1053→  enabled,
  1054→  onUpdated,
  1055→}: {
  1056→  open: boolean;
  1057→  onOpenChange: (v: boolean) => void;
  1058→  enabled: boolean;
  1059→  onUpdated: (enabled: boolean) => void;
  1060→}) {
  1061→  const isMobile = useIsMobile();
  1062→  const [step, setStep] = useState<"main" | "setup" | "verify" | "backup" | "disable">("main");
  1063→  const [secret, setSecret] = useState("");
  1064→  const [qrCode, setQrCode] = useState("");
  1065→  const [code, setCode] = useState("");
  1066→  const [backupCodes, setBackupCodes] = useState<string[]>([]);
  1067→  const [disablePassword, setDisablePassword] = useState("");
  1068→  const [loading, setLoading] = useState(false);
  1069→
  1070→  useEffect(() => {
  1071→    if (open) {
  1072→      // Reset the multi-step dialog each time a new interaction begins.
  1073→      // eslint-disable-next-line react-hooks/set-state-in-effect
  1074→      setStep("main");
  1075→      setSecret("");
  1076→      setQrCode("");
  1077→      setCode("");
  1078→      setBackupCodes([]);
  1079→      setDisablePassword("");
  1080→    }
  1081→  }, [open]);
  1082→
  1083→  const startSetup = async () => {
  1084→    setLoading(true);
  1085→    try {
  1086→      const res = await api.post<{ success: boolean; data: { secret: string; qrCode: string } }>("/auth/2fa/setup");
  1087→      setSecret(res.data.secret);
  1088→      setQrCode(res.data.qrCode);
  1089→      setStep("setup");
  1090→    } catch (e: unknown) {
  1091→      toast.error((e as Error).message || "Failed to start 2FA setup");
  1092→    } finally {
  1093→      setLoading(false);
  1094→    }
  1095→  };
  1096→
  1097→  const verifyAndEnable = async () => {
  1098→    if (!/^\d{6}$/.test(code)) {
  1099→      toast.error("Enter the 6-digit code");
  1100→      return;
  1101→    }
  1102→    setLoading(true);
  1103→    try {
  1104→      const res = await api.post<{ success: boolean; data: { backupCodes: string[] } }>("/auth/2fa/verify", {
  1105→        secret,
  1106→        code,
  1107→      });
  1108→      setBackupCodes(res.data.backupCodes);
  1109→      setStep("backup");
  1110→      toast.success("Two-factor authentication enabled!");
  1111→    } catch (e: unknown) {
  1112→      toast.error((e as Error).message || "Invalid code. Try again.");
  1113→    } finally {
  1114→      setLoading(false);
  1115→    }
  1116→  };
  1117→
  1118→  const disable2FA = async () => {
  1119→    setLoading(true);
  1120→    try {
  1121→      await api.post("/auth/2fa/disable", { password: disablePassword });
  1122→      onUpdated(false);
  1123→      toast.success("Two-factor authentication disabled");
  1124→      onOpenChange(false);
  1125→    } catch (e: unknown) {
  1126→      toast.error((e as Error).message || "Failed to disable 2FA");
  1127→    } finally {
  1128→      setLoading(false);
  1129→    }
  1130→  };
  1131→

  1358→function BackupRegenerator({
  1359→  onDone,
  1360→  onCancel,
  1361→}: {
  1362→  onDone: (codes: string[]) => void;
  1363→  onCancel: () => void;
  1364→}) {
  1365→  const [code, setCode] = useState("");
  1366→  const [loading, setLoading] = useState(false);
  1367→
  1368→  const regenerate = async () => {
  1369→    if (!/^\d{6}$/.test(code)) {
  1370→      toast.error("Enter the 6-digit code from your authenticator");
  1371→      return;
  1372→    }
  1373→    setLoading(true);
  1374→    try {
  1375→      const res = await api.post<{ success: boolean; data: { backupCodes: string[] } }>("/auth/2fa/backup-codes", { code });
  1376→      onDone(res.data.backupCodes);
  1377→      toast.success("New backup codes generated");
  1378→    } catch (e: unknown) {
  1379→      toast.error((e as Error).message || "Invalid code");
  1380→    } finally {
  1381→      setLoading(false);
  1382→    }
  1383→  };
  1384→
  1385→  return (
  1386→    <div className="space-y-4">
  1387→      <div className="text-center">
  1388→        <div className="grid place-items-center h-14 w-14 rounded-3xl bg-primary/15 text-primary mx-auto mb-3">
  1389→          <RefreshCw className="h-7 w-7" />
  1390→        </div>
  1391→        <h3 className="font-semibold text-lg">Regenerate Backup Codes</h3>
  1392→        <p className="text-xs text-muted-foreground mt-1">
  1393→          Enter a code from your authenticator app to generate fresh backup codes.
  1394→        </p>
  1395→      </div>
  1396→      <GlassInput
  1397→        label="6-digit code"
  1398→        value={code}
  1399→        onChange={(e) => setCode(e.target.value.replace(/\D/g, "").slice(0, 6))}
  1400→        placeholder="000000"
  1401→        icon={<KeyRound />}
  1402→        className="text-center text-lg tracking-[0.5em] font-mono"
  1403→      />
  1404→      <div className="flex gap-2">
  1405→        <GlassButton variant="ghost" className="flex-1" onClick={onCancel}>
  1406→          Cancel
  1407→        </GlassButton>
  1408→        <GlassButton className="flex-1" loading={loading} onClick={regenerate} disabled={code.length !== 6}>
  1409→          <RefreshCw className="h-4 w-4" />
  1410→          Generate
  1411→        </GlassButton>
  1412→      </div>
  1413→    </div>
  1414→  );
  1415→}
