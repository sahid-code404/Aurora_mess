    99→function isoDate(d: Date): string {
   100→  const y = d.getFullYear();
   101→  const m = String(d.getMonth() + 1).padStart(2, "0");
   102→  const day = String(d.getDate()).padStart(2, "0");
   103→  return `${y}-${m}-${day}`;
   104→}
   105→
   106→function isSameDay(a: Date, b: Date): boolean {
   107→  return (
   108→    a.getFullYear() === b.getFullYear() &&
   109→    a.getMonth() === b.getMonth() &&
   110→    a.getDate() === b.getDate()
   111→  );
   112→}
   113→
   114→function isPastDate(d: Date, today = new Date()): boolean {
   115→  const a = new Date(d.getFullYear(), d.getMonth(), d.getDate());
   116→  const b = new Date(today.getFullYear(), today.getMonth(), today.getDate());
   117→  return a.getTime() < b.getTime();
   118→}
   119→
   120→function relativeEditableUntil(iso: string): string {
   121→  const target = new Date(iso);
   122→  const now = new Date();
   123→  const diff = target.getTime() - now.getTime();
   124→  if (diff <= 0) return "Locked";
   125→  const mins = Math.floor(diff / 60000);
   126→  if (mins < 60) return `in ${mins}m`;
   127→  const hours = Math.floor(mins / 60);
   128→  const remMins = mins % 60;
   129→  if (hours < 24) return `in ${hours}h ${remMins}m`;
   130→  const days = Math.floor(hours / 24);
   131→  return `in ${days}d`;
   132→}

   134→function formatTimeRange(start: string, end: string): string {
   135→  const to12 = (t: string) => {
   136→    const [h, m] = t.split(":").map(Number);
   137→    const period = h >= 12 ? "PM" : "AM";
   138→    const hr = h % 12 === 0 ? 12 : h % 12;
   139→    return `${hr}:${String(m || 0).padStart(2, "0")} ${period}`;
   140→  };
   141→  return `${to12(start)} – ${to12(end)}`;
   142→}
   143→
   144→function formatEditableUntil(iso: string): string {
   145→  return new Date(iso).toLocaleString("en-US", {
   146→    weekday: "short",
   147→    day: "2-digit",
   148→    month: "short",
   149→    hour: "2-digit",
   150→    minute: "2-digit",
   151→    hour12: true,
   152→  });
   153→}

   415→function weekRangeLabel(d: Date): string {
   416→  const start = startOfWeek(d);
   417→  const end = endOfWeek(d);
   418→  const sameMonth = start.getMonth() === end.getMonth();
   419→  if (sameMonth) {
   420→    return `${MONTHS[start.getMonth()].slice(0, 3)} ${start.getDate()} – ${end.getDate()}, ${start.getFullYear()}`;
   421→  }
   422→  return `${MONTHS[start.getMonth()].slice(0, 3)} ${start.getDate()} – ${MONTHS[end.getMonth()].slice(0, 3)} ${end.getDate()}, ${end.getFullYear()}`;
   423→}
   424→
   425→function startOfWeek(d: Date): Date {
   426→  const r = new Date(d);
   427→  r.setDate(d.getDate() - d.getDay());
   428→  r.setHours(0, 0, 0, 0);
   429→  return r;
   430→}
   431→
   432→function endOfWeek(d: Date): Date {
   433→  const r = startOfWeek(d);
   434→  r.setDate(r.getDate() + 6);
   435→  return r;
   436→}

   442→function MonthView({
   443→  cursor,
   444→  byDate,
   445→  onToggle,
   446→}: {
   447→  cursor: Date;
   448→  byDate: Record<string, MealEntry[]>;
   449→  onToggle: (entry: MealEntry, next: "ON" | "OFF") => void;
   450→}) {
   451→  const today = new Date();
   452→  const year = cursor.getFullYear();
   453→  const month = cursor.getMonth();
   454→  const firstDay = new Date(year, month, 1);
   455→  const startOffset = firstDay.getDay();
   456→  const daysInMonth = new Date(year, month + 1, 0).getDate();
   457→  const totalCells = Math.ceil((startOffset + daysInMonth) / 7) * 7;
   458→
   459→  const cells: Array<{ date: Date | null; iso: string | null }> = [];
   460→  for (let i = 0; i < totalCells; i++) {
   461→    const dayNum = i - startOffset + 1;
   462→    if (dayNum < 1 || dayNum > daysInMonth) {
   463→      cells.push({ date: null, iso: null });
   464→    } else {
   465→      const d = new Date(year, month, dayNum);
   466→      cells.push({ date: d, iso: isoDate(d) });
   467→    }
   468→  }
   469→
   470→  return (
   471→    <GlassCard className="p-3 md:p-5" hover={false}>
   472→      {/* Weekday header */}
   473→      <div className="grid grid-cols-7 gap-1 mb-2">
   474→        {WEEKDAYS.map((d) => (
   475→          <div
   476→            key={d}
   477→            className="text-center text-[10px] md:text-xs font-semibold uppercase tracking-wider text-muted-foreground py-1"
   478→          >
   479→            {d.slice(0, 2)}
   480→          </div>
   481→        ))}
   482→      </div>
   483→
   484→      {/* Calendar grid */}
   485→      <div className="grid grid-cols-7 gap-1">
   486→        {cells.map((cell, i) => {
   487→          if (!cell.date || !cell.iso) {
   488→            return (
   489→              <div
   490→                key={`empty-${i}`}
   491→                className="rounded-xl min-h-16 md:min-h-24 bg-muted/10"
   492→              />
   493→            );
   494→          }
   495→          const entries = byDate[cell.iso] || [];
   496→          const isToday = isSameDay(cell.date, today);
   497→          const past = isPastDate(cell.date, today);
   498→          return (
   499→            <motion.div
   500→              key={cell.iso}
   501→              initial={{ opacity: 0, scale: 0.95 }}
   502→              animate={{ opacity: 1, scale: 1 }}
   503→              transition={{ delay: Math.min(i * 0.01, 0.2) }}
   504→              className={cn(
   505→                "rounded-xl p-1.5 min-h-16 md:min-h-24 flex flex-col gap-1 relative overflow-hidden",
   506→                isToday ? "ring-2 ring-primary/70 bg-primary/5" : "bg-muted/20",
   507→                past && "opacity-60"
   508→              )}
   509→            >
   510→              <div className="flex items-center justify-between">
   511→                <span
   512→                  className={cn(
   513→                    "text-[10px] md:text-xs font-semibold",
   514→                    isToday ? "text-primary" : "text-muted-foreground"
   515→                  )}
   516→                >
   517→                  {cell.date.getDate()}
   518→                </span>
   519→                {past && entries.length > 0 && (
   520→                  <Lock className="h-2.5 w-2.5 text-muted-foreground/70" />
   521→                )}
   522→              </div>
   523→              <div className="flex flex-col gap-0.5 flex-1 overflow-hidden">
   524→                {entries.slice(0, 3).map((e) => {
   525→                  const on = entryIsOn(e);
   526→                  return (
   527→                    <div
   528→                      key={e.id}
   529→                      title={`${e.mealDisplayName} • ${e.status}`}
   530→                      className={cn(
   531→                        "flex items-center gap-1 rounded-md px-1 py-0.5 text-[9px] md:text-[10px] font-medium truncate",
   532→                        on ? "text-white" : "text-muted-foreground"
   533→                      )}
   534→                      style={
   535→                        on
   536→                          ? { background: e.mealColor }
   537→                          : {
   538→                              background: `${e.mealColor}1A`,
   539→                              color: e.mealColor,
   540→                            }
   541→                      }
   542→                    >
   543→                      <span className="text-[10px]">{e.mealIcon}</span>
   544→                      <span className="truncate hidden sm:inline">
   545→                        {e.mealDisplayName}
   546→                      </span>
   547→                      {entryIsLocked(e) && (
   548→                        <Lock className="h-2 w-2 ml-auto shrink-0" />
   549→                      )}
   550→                    </div>
   551→                  );
   552→                })}
   553→                {entries.length > 3 && (
   554→                  <span className="text-[9px] md:text-[10px] text-muted-foreground px-1">
   555→                    +{entries.length - 3} more
   556→                  </span>
   557→                )}
   558→                {entries.length === 0 && (
   559→                  <span className="text-[9px] text-muted-foreground/40 px-1 hidden md:inline">
   560→                    —
   561→                  </span>
   562→                )}
   563→              </div>
   564→            </motion.div>
   565→          );
   566→        })}
   567→      </div>
   568→    </GlassCard>
   569→  );
   570→}

   572→// ─────────────────────────────────────────────────────────────

   576→function WeekView({
   577→  cursor,
   578→  byDate,
   579→  onToggle,
   580→}: {
   581→  cursor: Date;
   582→  byDate: Record<string, MealEntry[]>;
   583→  onToggle: (entry: MealEntry, next: "ON" | "OFF") => void;
   584→}) {
   585→  const today = new Date();
   586→  const start = startOfWeek(cursor);
   587→  const days = Array.from({ length: 7 }, (_, i) => {
   588→    const d = new Date(start);
   589→    d.setDate(start.getDate() + i);
   590→    return d;
   591→  });
   592→
   593→  return (
   594→    <div className="flex gap-3 overflow-x-auto no-scrollbar pb-2 md:grid md:grid-cols-7 md:overflow-visible">
   595→      {days.map((d) => {
   596→        const iso = isoDate(d);
   597→        const entries = byDate[iso] || [];
   598→        const isToday = isSameDay(d, today);
   599→        const past = isPastDate(d, today);
   600→        return (

   694→function AgendaView({
   695→  cursor,
   696→  byDate,
   697→  onToggle,
   698→}: {
   699→  cursor: Date;
   700→  byDate: Record<string, MealEntry[]>;
   701→  onToggle: (entry: MealEntry, next: "ON" | "OFF") => void;
   702→}) {
   703→  const today = new Date();
   704→  const year = cursor.getFullYear();
   705→  const month = cursor.getMonth();
   706→  const daysInMonth = new Date(year, month + 1, 0).getDate();
   707→
   708→  const items = Array.from({ length: daysInMonth }, (_, i) => {
   709→    const d = new Date(year, month, i + 1);
   710→    return { date: d, iso: isoDate(d), entries: byDate[isoDate(d)] || [] };
   711→  });
   712→
   713→  if (items.every((i) => i.entries.length === 0)) {
   714→    return (
   715→      <GlassCard className="p-10 md:p-14 text-center" hover={false}>
   716→        <div className="grid place-items-center gap-3">
   717→          <div className="grid place-items-center h-14 w-14 rounded-3xl bg-muted/40">
   718→            <CalendarDays className="h-7 w-7 text-muted-foreground" />
   719→          </div>
   720→          <p className="font-semibold text-lg">No meals scheduled</p>
   721→          <p className="text-sm text-muted-foreground max-w-sm">
   722→            No meals are configured for {MONTHS[month]} {year}. Contact your
   723→            administrator to set up meal configurations.
   724→          </p>
   725→        </div>
   726→      </GlassCard>
   727→    );
   728→  }
   729→
   730→  return (
   731→    <StaggerGroup className="space-y-3">
   732→      {items.map(({ date, iso, entries }) => {
   733→        const isToday = isSameDay(date, today);
   734→        const past = isPastDate(date, today);
   735→        return (
   736→          <StaggerItem key={iso}>
   737→            <div className="flex items-start gap-3">
   738→              {/* Date pill */}
   739→              <div
   740→                className={cn(
   741→                  "shrink-0 w-14 md:w-16 rounded-2xl py-2 px-1 text-center",
   742→                  isToday
   743→                    ? "bg-primary text-primary-foreground shadow-lg shadow-primary/30"
   744→                    : "glass-soft"
   745→                )}
   746→              >
   747→                <p className="text-[10px] font-semibold uppercase opacity-70">
   748→                  {WEEKDAYS[date.getDay()].slice(0, 3)}
   749→                </p>
   750→                <p className="text-xl md:text-2xl font-bold leading-tight">
   751→                  {date.getDate()}
   752→                </p>
   753→              </div>
   754→
   755→              {/* Entries */}
   756→              <div className="flex-1 min-w-0 space-y-2">
   757→                {entries.length === 0 ? (
   758→                  <div className="glass-soft rounded-2xl p-3 text-xs text-muted-foreground">
   759→                    No meals scheduled
   760→                  </div>
   761→                ) : (
   762→                  entries.map((e) => (
   763→                    <MealAgendaCard
   764→                      key={e.id}
   765→                      entry={e}
   766→                      onToggle={onToggle}
   767→                    />
   768→                  ))
   769→                )}
   770→              </div>
   771→            </div>
   772→          </StaggerItem>
   773→        );
   774→      })}
   775→    </StaggerGroup>
   776→  );
   777→}

   855→// ─────────────────────────────────────────────────────────────
   856→// Main CalendarView
   857→// ─────────────────────────────────────────────────────────────
   858→
   859→export function CalendarView() {
   860→  const isMobile = useIsMobile();
   861→  const qc = useQueryClient();
   862→  const [mode, setMode] = React.useState<ViewMode>("agenda");
   863→  const [cursor, setCursor] = React.useState<Date>(new Date());
   864→
   865→  // Mobile defaults to agenda, desktop defaults to month
   866→  React.useEffect(() => {
   867→    // Keep the interaction mode aligned with responsive layout changes.
   868→    // eslint-disable-next-line react-hooks/set-state-in-effect
   869→    setMode(isMobile ? "agenda" : "month");
   870→  }, [isMobile]);
   871→
   872→  const year = cursor.getFullYear();
   873→  const month = cursor.getMonth();
   874→
   875→  const queryKey = React.useMemo(
   876→    () => ["meals", "entries", year, month] as const,
   877→    [year, month]
   878→  );
   879→
   880→  const { data, isLoading, isError, error } = useQuery({
   881→    queryKey,
   882→    queryFn: async () => {
   883→      const r = await api.get<{ success: boolean; data: CalendarResponse }>(
   884→        "/meals/entries",
   885→        {
   886→          params: { year, month },
   887→        }
   888→      );
   889→      return r.data;
   890→    },
   891→    staleTime: 60_000,
   892→  });
   893→
   894→  // Toggle mutation with optimistic update
   895→  const toggleMutation = useMutation({
   896→    mutationFn: async ({
   897→      entryId,
   898→      status,
   899→    }: {
   900→      entryId: string;
   901→      status: "ON" | "OFF";
   902→    }) => {
   903→      return api.patch<MealEntry>("/meals/toggle", { entryId, status });
   904→    },
   905→    onMutate: async ({ entryId, status }) => {
   906→      // Cancel outgoing refetches
   907→      await qc.cancelQueries({ queryKey });
   908→
   909→      // Snapshot previous value
   910→      const previous = qc.getQueryData<CalendarResponse>(queryKey);
   911→
   912→      // Optimistically update
   913→      qc.setQueryData<CalendarResponse>(queryKey, (old) => {
   914→        if (!old) return old;
   915→        const newByDate: Record<string, MealEntry[]> = {};
   916→        for (const [key, entries] of Object.entries(old.byDate)) {
   917→          newByDate[key] = entries.map((e) =>
   918→            e.id === entryId
   919→              ? {
   920→                  ...e,
   921→                  status: e.locked ? "LOCKED" : status,
   922→                }
   923→              : e
   924→          );
   925→        }
   926→        return { ...old, byDate: newByDate };
   927→      });
   928→
   929→      return { previous };
   930→    },
   931→    onError: (err, _vars, ctx) => {
   932→      // Revert on error
   933→      if (ctx?.previous) {
   934→        qc.setQueryData(queryKey, ctx.previous);
   935→      }
   936→      const message =
   937→        err instanceof ApiError
   938→          ? err.message
   939→          : "Failed to update meal. Please try again.";
   940→      toast.error(message);
   941→    },
   942→    onSettled: () => {
   943→      qc.invalidateQueries({ queryKey: ["meals", "entries"] });
   944→    },
   945→  });
   946→
   947→  const handleToggle = React.useCallback(
   948→    (entry: MealEntry, next: "ON" | "OFF") => {
   949→      if (entryIsLocked(entry)) {
   950→        toast.error("This meal is locked and cannot be changed.");
   951→        return;
   952→      }
   953→      if (entry.status === next) return;
   954→      toggleMutation.mutate({ entryId: entry.id, status: next });
   955→    },
   956→    [toggleMutation]
   957→  );
   958→
   959→  const goPrev = React.useCallback(() => {
   960→    if (mode === "week") {
   961→      const d = new Date(cursor);
   962→      d.setDate(d.getDate() - 7);
   963→      setCursor(d);
   964→    } else {
   965→      setCursor(new Date(year, month - 1, 1));
   966→    }
   967→  }, [mode, cursor, year, month]);
   968→
   969→  const goNext = React.useCallback(() => {
   970→    if (mode === "week") {
   971→      const d = new Date(cursor);
   972→      d.setDate(d.getDate() + 7);
   973→      setCursor(d);
   974→    } else {
   975→      setCursor(new Date(year, month + 1, 1));
   976→    }
   977→  }, [mode, cursor, year, month]);
   978→
   979→  const goToday = React.useCallback(() => setCursor(new Date()), []);
   980→
   981→  const pickMonth = React.useCallback(
   982→    (y: number, m: number) => setCursor(new Date(y, m, 1)),
   983→    []
   984→  );
   985→
   986→  const byDate = data?.byDate || {};
   987→  const mealsCount = Object.values(byDate).reduce(
   988→    (s, arr) => s + arr.length,
   989→    0
   990→  );
   991→
   992→  return (
   993→    <StaggerGroup className="space-y-4 md:space-y-5 pb-6">
   994→      {/* View toggle */}
   995→      <StaggerItem>
   996→        <div className="flex items-center justify-end">
   997→          <GlassNav<ViewMode>
   998→            value={mode}
   999→            onChange={setMode}
  1000→            items={[
  1001→              {
  1002→                value: "agenda",
  1003→                label: "Agenda",
  1004→                icon: <List className="h-3.5 w-3.5" />,
  1005→              },
  1006→              {
  1007→                value: "week",
  1008→                label: "Week",
  1009→                icon: <Columns3 className="h-3.5 w-3.5" />,
  1010→              },
  1011→              {
  1012→                value: "month",
  1013→                label: "Month",
  1014→                icon: <Grid3x3 className="h-3.5 w-3.5" />,
  1015→              },
  1016→            ]}
  1017→          />
  1018→        </div>
  1019→      </StaggerItem>
  1020→
  1021→      {/* Quick nav */}
  1022→      <StaggerItem>
  1023→        <GlassCard className="p-3 md:p-4" hover={false}>
  1024→          <CalendarHeader
  1025→            view={mode}
  1026→            cursor={cursor}
  1027→            onPrev={goPrev}
  1028→            onNext={goNext}
  1029→            onToday={goToday}
  1030→            onPickMonth={pickMonth}
  1031→          />
  1032→        </GlassCard>
  1033→      </StaggerItem>
  1034→
  1035→      {/* Content */}
  1036→      <StaggerItem>
  1037→        <AnimatePresence mode="wait">
  1038→          {isLoading ? (
  1039→            <motion.div
  1040→              key="loading"
  1041→              initial={{ opacity: 0 }}
  1042→              animate={{ opacity: 1 }}
  1043→              exit={{ opacity: 0 }}
  1044→            >
  1045→              <CalendarSkeleton mode={mode} />
  1046→            </motion.div>
  1047→          ) : isError ? (
  1048→            <motion.div
  1049→              key="error"
  1050→              initial={{ opacity: 0 }}
  1051→              animate={{ opacity: 1 }}
  1052→              exit={{ opacity: 0 }}
  1053→            >
  1054→              <GlassCard className="p-8 md:p-12 text-center" hover={false}>
  1055→                <div className="grid place-items-center gap-3">
  1056→                  <div className="grid place-items-center h-14 w-14 rounded-3xl bg-destructive/15">
  1057→                    <AlertCircle className="h-7 w-7 text-destructive" />
  1058→                  </div>
  1059→                  <p className="font-semibold text-lg">
  1060→                    Couldn&apos;t load your calendar
  1061→                  </p>
  1062→                  <p className="text-sm text-muted-foreground max-w-sm">
  1063→                    {error instanceof ApiError
  1064→                      ? error.message
  1065→                      : "Something went wrong. Please try again."}
  1066→                  </p>
  1067→                  <GlassButton
  1068→                    variant="secondary"
  1069→                    size="sm"
  1070→                    onClick={() =>
  1071→                      qc.invalidateQueries({ queryKey: ["meals", "entries"] })
  1072→                    }
  1073→                  >
  1074→                    Retry
  1075→                  </GlassButton>
  1076→                </div>
  1077→              </GlassCard>
  1078→            </motion.div>
  1079→          ) : mealsCount === 0 ? (
  1080→            <motion.div
  1081→              key="empty"
  1082→              initial={{ opacity: 0 }}
  1083→              animate={{ opacity: 1 }}
  1084→              exit={{ opacity: 0 }}
  1085→            >
  1086→              <GlassCard className="p-10 md:p-14 text-center" hover={false}>
  1087→                <div className="grid place-items-center gap-3">
  1088→                  <div className="grid place-items-center h-14 w-14 rounded-3xl bg-muted/40">
  1089→                    <CalendarDays className="h-7 w-7 text-muted-foreground" />
  1090→                  </div>
  1091→                  <p className="font-semibold text-lg">No meals configured</p>
  1092→                  <p className="text-sm text-muted-foreground max-w-md">
  1093→                    Your administrator hasn&apos;t set up any meals for{" "}
  1094→                    {MONTHS[month]} {year}. Once meals are configured,
  1095→                    they&apos;ll appear here automatically.
  1096→                  </p>
  1097→                </div>
  1098→              </GlassCard>
  1099→            </motion.div>
  1100→          ) : (
  1101→            <motion.div
  1102→              key={mode}
  1103→              initial={{ opacity: 0, y: 8 }}
  1104→              animate={{ opacity: 1, y: 0 }}
  1105→              exit={{ opacity: 0, y: -8 }}
  1106→              transition={{ duration: 0.25 }}
  1107→            >
  1108→              {mode === "month" && (
  1109→                <MonthView
  1110→                  cursor={cursor}
  1111→                  byDate={byDate}
  1112→                  onToggle={handleToggle}
  1113→                />
  1114→              )}
  1115→              {mode === "week" && (
  1116→                <WeekView
  1117→                  cursor={cursor}
  1118→                  byDate={byDate}
  1119→                  onToggle={handleToggle}
  1120→                />
  1121→              )}
  1122→              {mode === "agenda" && (
  1123→                <AgendaView
  1124→                  cursor={cursor}
  1125→                  byDate={byDate}
  1126→                  onToggle={handleToggle}
  1127→                />
  1128→              )}
  1129→            </motion.div>
  1130→          )}
  1131→        </AnimatePresence>
  1132→      </StaggerItem>
  1133→
  1134→      {/* Footer: Legend + cutoff info */}
  1135→      {mealsCount > 0 && (
  1136→        <StaggerItem>
  1137→          <GlassCard className="p-3 md:p-4" hover={false}>
  1138→            <div className="flex flex-wrap items-center justify-between gap-3">
  1139→              <Legend />
  1140→              <p className="text-[11px] text-muted-foreground inline-flex items-center gap-1.5">
  1141→                <Info className="h-3 w-3" />
  1142→                Cutoff shown as{" "}
  1143→                <span className="font-medium text-foreground">
  1144→                  Editable [time]

  1146→                — meals lock automatically past that time.
  1147→              </p>
  1148→            </div>
  1149→          </GlassCard>
  1150→        </StaggerItem>
  1151→      )}
  1152→    </StaggerGroup>
  1153→  );
