    87→  const [acting, setActing] = useState(false);

   106→  if (isLoading) {

   108→      <div className="space-y-6">
   109→        <PageHeader title="Resident 360°" subtitle="One resident, every dimension." />

   111→      </div>

   113→  }
   114→

   117→      <div className="space-y-6">
   118→        <PageHeader title="Resident 360°" subtitle="One resident, every dimension." />

   125→          <GlassButton variant="secondary" icon={<ArrowLeft />} onClick={() => goBack("#/admin/residents")}>

   160→          className="mt-1 size-11 shrink-0 px-0"

   371→                  icon={Wallet}
   372→                  title={`${p.displayNumber} · ${p.method}`}

   377→                      <Money minor={p.amountMinor} className="text-sm font-semibold" />

   384→          )}

   390→        <GlassCard className="space-y-2 p-4 sm:p-5">

   393→            <EmptyState icon={ReceiptText} title="No bills yet" description="Bills appear after each billing period is generated." />

   415→
   416→      {/* --------------------------------- Tasks -------------------------------- */}
   417→      {tab === "tasks" && (
   418→        <GlassCard className="space-y-2 p-4 sm:p-5">

   423→            <div className="mt-3 space-y-1.5">
   424→              {tasks.map((t) => (
   425→                <ActivityItem
   426→                  key={t.id}
   427→                  icon={ClipboardList}
   428→                  title={t.description}
   429→                  description={t.dueDate ? `Due ${fmtDate(t.dueDate)}` : "No due date"}

   432→                    <span className="flex flex-col items-end gap-1">

   440→          )}
   441→        </GlassCard>
   442→      )}
   443→
   444→      {/* -------------------------------- Activity ------------------------------- */}
   445→      {tab === "activity" && (
   446→        <GlassCard className="space-y-2 p-4 sm:p-5">
   447→          <SectionHeading>Activity</SectionHeading>
   448→          {(data.audit as unknown[]).length === 0 ? (
   449→            <EmptyState icon={History} title="No activity recorded" description="Audit events for this resident will appear here." />

   451→            <p className="mt-3 text-[13px] text-muted-foreground">See the full Audit Trail for details.</p>

   491→/* ------------------------------------------------------------- membership */
   492→
   493→function toInputDate(iso: string | null): string {
   494→  if (!iso) return "";
   495→  const d = new Date(iso);
   496→  if (Number.isNaN(d.getTime())) return "";
   497→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
   498→}
   499→
   500→function MembershipDialog({
   501→  open,
   502→  onOpenChange,
   503→  residentId,
   504→  from,
   505→  until,
   506→  onSaved,
   507→}: {
   508→  open: boolean;
   509→  onOpenChange: (open: boolean) => void;
   510→  residentId: string;
   511→  from: string | null;
   512→  until: string | null;
   513→  onSaved: () => void;
   514→}) {
   515→  const [fromDate, setFromDate] = useState(() => toInputDate(from));
   516→  const [untilDate, setUntilDate] = useState(() => toInputDate(until));
   517→  const [saving, setSaving] = useState(false);
   518→
   519→  async function save() {
   520→    setSaving(true);
   521→    try {
   522→      await patchJson(`/api/v1/admin/residents/${residentId}/membership`, {
   523→        membershipEffectiveFrom: fromDate ? new Date(`${fromDate}T00:00:00`).toISOString() : null,
   524→        membershipEffectiveUntil: untilDate ? new Date(`${untilDate}T23:59:59`).toISOString() : null,
   525→      });
   526→      toast.success("Membership updated", {
   527→        description: "Meal availability follows the new dates from the next evaluation.",
   528→      });
   529→      onSaved();
   530→      onOpenChange(false);
   531→    } catch (err) {
   532→      // The backend surfaces the closed-period restriction with plain language.
   533→      toast.error(errMessage(err));
   534→    } finally {
   535→      setSaving(false);
   536→    }
   537→  }
   538→
   539→  return (
   540→    <DetailDialog
   541→      open={open}
   542→      onOpenChange={onOpenChange}
   543→      title="Edit membership dates"
   544→      description="Changing the start date into an already-billed period is restricted — meals and charges for closed periods stay fixed."
   545→      footer={
   546→        <>
   547→          <GlassButton variant="ghost" onClick={() => onOpenChange(false)} disabled={saving}>
   548→            Cancel
   549→          </GlassButton>
   550→          <GlassButton loading={saving} onClick={() => void save()}>
   551→            Save membership
   552→          </GlassButton>
   553→        </>
   554→      }
   555→    >
   556→      <div className="space-y-4">
   557→        <TextField label="Member from" type="date" value={fromDate} onChange={setFromDate} />
   558→        <TextField label="Member until (optional)" type="date" value={untilDate} onChange={setUntilDate} hint="Leave empty for an open-ended membership." />
   559→      </div>
   560→    </DetailDialog>
   561→  );
   562→}

   566→function ExemptionDialog({
   567→  open,
   568→  onOpenChange,
   569→  residentId,
   570→  residentName,
   571→  onSaved,
   572→}: {
   573→  open: boolean;
   574→  onOpenChange: (open: boolean) => void;
   575→  residentId: string;
   576→  residentName: string;
   577→  onSaved: () => void;
   578→}) {
   579→  const [reason, setReason] = useState("");
   580→  const [expiresAt, setExpiresAt] = useState("");
   581→  const [saving, setSaving] = useState(false);
   582→
   583→  async function save() {
   584→    setSaving(true);
   585→    try {
   586→      await postJson("/api/v1/admin/policy-exemptions", {
   587→        residentId,
   588→        reason: reason.trim(),
   589→        expiresAt: expiresAt || undefined,
   590→      });
   591→      toast.success("Exemption created", { description: `${residentName} is exempt until ${expiresAt || "cancelled"}.` });
   592→      setReason("");
   593→      setExpiresAt("");
   594→      onSaved();
   595→      onOpenChange(false);
   596→    } catch (err) {
   597→      toast.error(errMessage(err));
   598→    } finally {
   599→      setSaving(false);
   600→    }
   601→  }
   602→
   603→  const valid = reason.trim().length >= 3;
   604→
   605→  return (
   606→    <DetailDialog
   607→      open={open}
   608→      onOpenChange={onOpenChange}
   609→      title="Create deficit-policy exemption"
   610→      description="While active, deficit restrictions (meal limiting) are lifted for this resident. Audited with your reason."
   611→      footer={
   612→        <>
   613→          <GlassButton variant="ghost" onClick={() => onOpenChange(false)} disabled={saving}>
   614→            Cancel
   615→          </GlassButton>
   616→          <GlassButton loading={saving} disabled={!valid} onClick={() => void save()}>
   617→            Create exemption
   618→          </GlassButton>
   619→        </>
   620→      }
   621→    >
   622→      <div className="space-y-4">
   623→        <TextField
   624→          label="Reason"
   625→          value={reason}
   626→          onChange={setReason}
   627→          maxLength={500}
   628→          placeholder="e.g. Salary delayed — settling by month end"
   629→        />
   630→        <TextField
   631→          label="Expires on (optional)"
   632→          type="date"
   633→          value={expiresAt}
   634→          onChange={setExpiresAt}
   635→          hint="Leave empty to keep the exemption until cancelled."
   636→        />
   637→        <div className="glass-inset flex items-center justify-between rounded-md px-3.5 py-2.5">
   638→          <span className="text-xs font-semibold text-muted-foreground">Policy</span>
   639→          <Chip tone="frost">Deficit restriction</Chip>
   640→        </div>
   641→      </div>
   642→    </DetailDialog>
   643→  );
   644→}
