    97→   152→async function unwrap<T>(promise: Promise<unknown>): Promise<T> {
    98→   153→  const res = await promise;
    99→   154→  if (res && typeof res === "object" && "success" in res && "data" in (res as Record<string, unknown>)) {
   100→   155→    return (res as unknown as { data: T }).data;
   101→   156→  }
   102→   157→  return res as T;
   103→   158→}
