    81→    81→/** (year, monthIndex 0-11) → "YYYY-MM". */
    82→    82→function monthKeyOf(year: number, month: number): string {
    83→    83→  return `${year}-${String(month + 1).padStart(2, "0")}`;
    84→    84→}
    85→    85→
    86→    86→function monthStartKey(year: number, month: number): string {
    87→    87→  return `${monthKeyOf(year, month)}-01`;
    88→    88→}
    89→    89→
    90→    90→function monthEndKey(year: number, month: number): string {
    91→    91→  const lastDay = new Date(Date.UTC(year, month + 1, 0)).getUTCDate();
    92→    92→  return `${monthKeyOf(year, month)}-${String(lastDay).padStart(2, "0")}`;
    93→    93→}

    96→    96→function monthLongLabel(year: number, month: number): string {
    97→    97→  return new Intl.DateTimeFormat("en-IN", { month: "long", year: "numeric", timeZone: "UTC" }).format(
    98→    98→    new Date(Date.UTC(year, month, 1))
    99→    99→  );
   100→   100→}

   109→   109→/** Picker pill labels — relative for ±1 day, compact otherwise (BoardOps pattern). */
   110→   110→function dayPillLabels(key: string, todayKey: string): { top: string; bottom: string } {
   111→   111→  if (key === todayKey) return { top: "Today", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   112→   112→  if (key === addDaysToKey(todayKey, -1)) return { top: "Yesterday", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   113→   113→  if (key === addDaysToKey(todayKey, 1)) return { top: "Tomorrow", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   114→   114→  return { top: formatDateInTz(key, "UTC"), bottom: weekdayShort(key) };
   115→   115→}

   216→   216→  /* --------------------------- optimistic toggle --------------------------- */
   217→   217→
   218→   218→  async function handleToggle(meal: MealInstanceDto, next: "ON" | "OFF") {
   219→   219→    const queries = queryClient.getQueriesData<MealsCacheValue>({
   220→   220→      queryKey: ["api", "/api/v1/meals"],
   221→   221→    });
   222→   222→    const snapshots = queries.map(([key, value]) => ({ key, value }));
   223→   223→
   224→   224→    let current: MealInstanceDto | undefined;
   225→   225→    for (const { value } of snapshots) {
   226→   226→      current = value?.data.find((m) => m.id === meal.id);
   227→   227→      if (current) break;
   228→   228→    }
   229→   229→    if (!current) return;
   230→   230→
   231→   231→    const expectedVersion = current.myState.version;
   232→   232→
   233→   233→    // Optimistically flip every cached copy of this row.
   234→   234→    for (const { key } of snapshots) {
   235→   235→      queryClient.setQueryData<MealsCacheValue>(key, (old) =>
   236→   236→        patchCacheRows(old, meal.id, (m) => ({
   237→   237→          ...m,
   238→   238→          myState: { ...m.myState, effectiveState: next } as MealMyState,
   239→   239→        }))
   240→   240→      );
   241→   241→    }
   242→   242→
   243→   243→    try {
   244→   244→      const res = await apiJson<ToggleResponse>(`/api/v1/meals/${meal.id}/toggle`, "POST", {
   245→   245→        state: next,
   246→   246→        expectedVersion,
   247→   247→      });
   248→   248→      // Authoritative response wins.
   249→   249→      for (const { key } of snapshots) {
   250→   250→        queryClient.setQueryData<MealsCacheValue>(key, (old) =>
   251→   251→          patchCacheRows(old, meal.id, (m) => ({
   252→   252→            ...m,
   253→   253→            myState: {
   254→   254→              ...m.myState,
   255→   255→              effectiveState: res.state,
   256→   256→              effectiveReason: res.effectiveReason,
   257→   257→              locked: res.locked,
   258→   258→              version: res.version,
   259→   259→            } as MealMyState,
   260→   260→          }))
   261→   261→        );
   262→   262→      }
   263→   263→      invalidate([RESIDENT_KEYS.dashboard]);
   264→   264→    } catch (err) {
   265→   265→      // Rollback — the toggle springs back (spec §114).
   266→   266→      for (const { key, value } of snapshots) {
   267→   267→        queryClient.setQueryData<MealsCacheValue>(key, value);
   268→   268→      }
   269→   269→      if (err instanceof ApiClientError) {
   270→   270→        if (err.code === "MEAL_CUTOFF_PASSED") {
   271→   271→          flashFor(meal.id, {
   272→   272→            tone: "warning",
   273→   273→            text: `${meal.name} is already locked. Admin can still override it.`,
   274→   274→          });
   275→   275→        } else if (err.code === "RESOURCE_CHANGED") {
   276→   276→          flashFor(meal.id, { tone: "info", text: "This meal was just changed. Refreshing…" });
   277→   277→          void queryClient.invalidateQueries({ queryKey: ["api", "/api/v1/meals"] });
   278→   278→        } else if (err.code === "MEAL_NOT_AVAILABLE") {
   279→   279→          flashFor(meal.id, { tone: "warning", text: "This meal isn't available to change right now." });
   280→   280→        } else {
   281→   281→          flashFor(meal.id, { tone: "danger", text: err.message });
   282→   282→        }
   283→   283→      } else {
   284→   284→        flashFor(meal.id, { tone: "danger", text: friendlyError(err) });
   285→   285→      }
   286→   286→    }
   287→   287→  }
