    52→interface DraftItem {
    53→  key: string;
    54→  itemName: string;
    55→  quantity: string;
    56→  unit: string;
    57→  unitPrice: string;
    58→}
    59→
    60→function draftItem(): DraftItem {
    61→  return { key: crypto.randomUUID(), itemName: "", quantity: "1", unit: "kg", unitPrice: "" };
    62→}
    63→
    64→function itemEstimateMinor(item: DraftItem): number | null {
    65→  const price = parseMoneyToMinor(item.unitPrice);
    66→  if (price == null) return null;
    67→  const qty = Number(item.quantity);
    68→  if (!Number.isFinite(qty) || qty <= 0) return null;
    69→  // Same rounding as the server: round-half-up on each line.
    70→  return Math.round(qty * price);
    71→}

   102→        continue;

   109→  const valid =

   390→/** "2025-09" ± 1 → "2025-08" / "2025-10". */
   391→function shiftMonthKey(key: string, delta: number): string {
   392→  const [y, m] = key.split("-").map(Number);
   393→  const d = new Date(y ?? 2025, (m ?? 1) - 1 + delta, 1);
   394→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
   395→}
   396→
   397→/** Long month name for the picker pill ("September"). */
   398→function monthLongName(key: string): string {
   399→  const [y, m] = key.split("-").map(Number);
   400→  return new Intl.DateTimeFormat("en-US", { month: "long" }).format(new Date(y ?? 2025, (m ?? 1) - 1, 1));
   401→}
   402→
   403→/** "Sep 2025" for the month-scoped KPI label. */
   404→function monthShortLabel(key: string): string {
   405→  return monthLabel(Number(key.slice(0, 4)), Number(key.slice(5, 7)));
   406→}

   408→export default function AdminExpenses() {
   409→  const [status, setStatus] = useState("ALL");
   410→  // Month scope — undefined = the server's current month (institution tz).
   411→  // Navigating sets an explicit YYYY-MM; the list AND the month KPIs follow it
   412→  // (the route scopes both to the requested month and echoes it in meta.month).
   413→  const [monthParam, setMonthParam] = useState<string | undefined>(undefined);
   414→  const [formOpen, setFormOpen] = useState(false);
   415→  const [detail, setDetail] = useState<ExpenseRow | null>(null);
   416→  const [action, setAction] = useState<"approve" | "reject" | "void" | null>(null);
   417→  const [acting, setActing] = useState(false);
   418→  const invalidate = useInvalidate();
   419→  const { institution } = useSession();
   420→  const tz = institution?.timezone ?? "Asia/Kolkata";
   421→
   422→  const { data: envelope, isLoading, error, refetch } = useApiMetaQuery<ExpenseRow[]>(EXPENSES_PATH, {
   423→    status: status === "ALL" ? undefined : status,
   424→    month: monthParam,
   425→  });
   426→  const expenses = envelope?.data ?? [];
   427→  const meta = envelope?.meta ?? {};
   428→
   429→  const categoriesQuery = useApiQuery<ExpenseCategory[]>(formOpen ? CATEGORIES_PATH : null);
   430→  const categories = categoriesQuery.data ?? [];
   431→
   432→  const pendingCount = expenses.filter((e) => e.status === "PENDING").length;
   433→
   434→  // Effective month — the server echoes the resolved month in meta (it owns
   435→  // the institution timezone); fall back to the client month before load.
   436→  const thisMonthKey = todayKey().slice(0, 7);
   437→  const activeMonthKey = metaStr(meta, "month") ?? monthParam ?? thisMonthKey;
   438→  const isThisMonth = activeMonthKey === thisMonthKey;
   439→
   440→  async function runAction(kind: "approve" | "reject" | "void", reason?: string) {
   441→    if (!detail) return;
   442→    setActing(true);
   443→    try {
   444→      await postJson(`${EXPENSES_PATH}/${detail.id}/${kind}`, kind === "approve" ? {} : { reason });
   445→      invalidate([EXPENSES_PATH, "/api/v1/admin/funds", "/api/v1/admin/dashboard", "/api/v1/admin/billing"]);
   446→      toast.success(kind === "approve" ? "Expense approved" : kind === "reject" ? "Expense rejected" : "Expense voided", {
   447→        description: `${detail.displayNumber} · ${detail.totalFormatted}`,
   448→      });
   449→      setAction(null);
   450→      setDetail(null);
   451→    } catch (err) {
   452→      toast.error(errMessage(err));
   453→    } finally {
   454→      setActing(false);
   455→    }
   456→  }
   457→
   458→  if (error) {
   459→    return (
   460→      <div className="space-y-4">
   461→        <ErrorState
   462→          code={(error as ApiClientError | undefined)?.code}
   463→          message={(error as ApiClientError | undefined)?.message}
   464→          onRetry={() => void refetch()}
   465→        />
   466→      </div>
   467→    );
   468→  }
   469→
   470→  return (
   471→    <div className="space-y-4">
   472→      {/* Month capsule — circular arrows + reset pill (BoardOps picker) */}
   473→      <PickerCapsule
   474→        onPrev={() => setMonthParam(shiftMonthKey(activeMonthKey, -1))}
   475→        onNext={() => setMonthParam(shiftMonthKey(activeMonthKey, 1))}
   476→        prevLabel="Previous month"
   477→        nextLabel="Next month"
   478→        onPillClick={() => setMonthParam(undefined)}
   479→        pillAriaLabel="Reset to the current month"
   480→        resettable={!isThisMonth}
   481→      >
   482→        <Calendar className="size-4 shrink-0 text-primary" aria-hidden />
   483→        <span className="min-w-0 text-center leading-tight">
   484→          <span className="block truncate text-sm font-bold text-primary">{monthLongName(activeMonthKey)}</span>
   485→          <span className="block truncate text-[11px] text-muted-foreground">{activeMonthKey.slice(0, 4)}</span>
   486→        </span>
   487→      </PickerCapsule>
   488→
   489→      <KpiGrid
   490→        loading={isLoading && !envelope}
   491→        className="lg:grid-cols-3"
   492→        kpis={[
   493→          {
   494→            label: `Total Expenses · ${monthShortLabel(activeMonthKey)}`,
   495→            value: metaStr(meta, "expensesThisMonthFormatted") ?? "—",
   496→            icon: <ReceiptText />,
   497→            tone: "primary",
   498→            glow: "primary",
   499→            sub: `${metaNum(meta, "entriesThisMonth") ?? 0} item entries approved`,
   500→          },
   501→          {
   502→            label: "Remaining Funds",
   503→            value: metaStr(meta, "remainingFundsFormatted") ?? "—",
   504→            icon: <Package />,
   505→            tone: "success",
   506→            glow: "success",
   507→            sub: "Cash & bank balance",
   508→          },
   509→          {
   510→            label: "Awaiting Review",
   511→            value: String(pendingCount),
   512→            icon: <FileText />,
   513→            tone: "warning",
   514→            glow: "warning",
   515→            sub: "Pending expenses",
   516→          },
   517→        ]}
   518→      />
   519→
   520→      <FilterChips
   521→        chips={[
   522→          { value: "ALL", label: "All" },
   523→          { value: "PENDING", label: "Pending" },
   524→          { value: "APPROVED", label: "Approved" },
   525→          { value: "REJECTED", label: "Rejected" },
   526→          { value: "VOIDED", label: "Voided" },
   527→        ]}
   528→        value={status}
   529→        onChange={setStatus}
   530→      />
   531→
   532→      {/* Primary action — BoardOps action bar, right-aligned above the list */}
   533→      <div className="flex flex-wrap items-center justify-end gap-2">
   534→        <GlassButton variant="primary" icon={<Plus />} onClick={() => setFormOpen(true)}>
   535→          Add Expense
   536→        </GlassButton>
   537→      </div>
   538→
   539→      {isLoading && !envelope ? (
   540→        <ListSkeleton rows={5} />
   541→      ) : expenses.length === 0 ? (
   542→        <EmptyState
   543→          icon={ReceiptText}
   544→          title={status === "PENDING" ? "No pending expenses" : "No expenses recorded yet"}
   545→          description={
   546→            status === "PENDING"
   547→              ? "Recorded expenses waiting for your approval will appear here."
   548→              : "Use Add Expense to record the first mess purchase."
   549→          }
   550→          action={
   551→            status !== "PENDING" ? (
   552→              <GlassButton variant="secondary" icon={<Plus />} onClick={() => setFormOpen(true)}>
   553→                Add Expense
   554→              </GlassButton>
   555→            ) : undefined
   556→          }
   557→        />
   558→      ) : (
   559→        <div className="max-h-[28rem] space-y-2.5 overflow-y-auto pr-1">
   560→          <AnimatePresence mode="popLayout">
   561→            {expenses.map((e, i) => (
   562→              <motion.div
   563→                key={e.id}
   564→                initial={{ opacity: 0, y: 6 }}
   565→                animate={{ opacity: 1, y: 0 }}
   566→                exit={{ opacity: 0, y: -4, scale: 0.98, transition: { duration: 0.15 } }}
   567→                transition={{ ...SPRING_SNAPPY, delay: Math.min(i * 0.04, 0.2) }}
   568→              >
   569→                <GlassCard className="p-4">
   570→                  <div className="flex items-start gap-3">
   571→                    {(() => {
   572→                      const tile = expenseTile(e.status);
   573→                      const TileIcon = tile.Icon;
   574→                      return (
   575→                        <span
   576→                          aria-hidden
   577→                          className={cn(
   578→                            "flex size-10 shrink-0 items-center justify-center rounded-md border [&_svg]:size-5",
   579→                            tile.className
   580→                          )}
   581→                        >
   582→                          <TileIcon aria-hidden />
   583→                        </span>
   584→                      );
   585→                    })()}
   586→                    <div className="min-w-0 flex-1">
   587→                      <div className="flex flex-wrap items-center gap-2">
   588→                        <p className="min-w-0 truncate text-sm font-medium" title={e.description}>
   589→                          {e.description}
   590→                        </p>
   591→                        <StatusBadge status={e.status} />
   592→                        {e.source === "TASK" && <Chip tone="frost">via task</Chip>}
   593→                      </div>
   594→                      <p className="kpi-num mt-0.5 truncate text-[11px] text-muted-foreground">
   595→                        {e.displayNumber} · {e.categoryName ?? "No category"} · {e.dateKey ? fmtDate(e.dateKey) : "—"} ·{" "}
   596→                        {e.itemCount ?? 0} item{(e.itemCount ?? 0) === 1 ? "" : "s"}
   597→                      </p>
   598→                      <div className="mt-2 flex flex-wrap items-baseline gap-x-4 gap-y-1">
   599→                        <div className="flex items-baseline gap-1">
   600→                          <span className="text-[10px] font-medium uppercase tracking-wide text-muted-foreground">
   601→                            Total
   602→                          </span>
   603→                          <Money minor={e.totalMinor} className="text-base font-bold" />
   604→                        </div>
   605→                        <span className="kpi-num text-[11px] text-muted-foreground/80">
   606→                          Recorded {fmtDateTime(e.createdAt, tz)}
   607→                        </span>
   608→                      </div>
   609→                    </div>
   610→                    <OverflowMenu
   611→                      actions={[
   612→                        { key: "view", label: "View details", icon: <FileText />, onSelect: () => setDetail(e) },
   613→                        ...(e.status === "PENDING"
   614→                          ? [
   615→                              { key: "approve", label: "Approve", icon: <CheckCircle2 />, onSelect: () => setDetail(e), separatorBefore: true },
   616→                              { key: "reject", label: "Reject…", icon: <XCircle />, onSelect: () => setDetail(e) },
   617→                            ]
   618→                          : []),
   619→                        ...(e.status === "APPROVED"
   620→                          ? [{ key: "void", label: "Void…", icon: <Trash2 />, onSelect: () => setDetail(e), destructive: true, separatorBefore: true }]
   621→                          : []),
   622→                      ]}
   623→                    />
   624→                  </div>
   625→                </GlassCard>
   626→              </motion.div>
   627→            ))}
   628→          </AnimatePresence>
   629→        </div>
   630→      )}
   631→
   632→      <ExpenseFormDialog
   633→        open={formOpen}
   634→        onOpenChange={setFormOpen}
   635→        categories={categories}
   636→        onSaved={() => invalidate([EXPENSES_PATH, "/api/v1/admin/dashboard"])}
   637→      />
   638→
   639→      <ExpenseDetailDialog
   640→        expense={detail}
   641→        tz={tz}
   642→        onClose={() => setDetail(null)}
   643→        onAction={(kind) => setAction(kind)}
   644→      />
   645→
   646→      {action && detail && (
   647→        <ConfirmDialog
   648→          open
   649→          onOpenChange={(open) => !open && setAction(null)}
   650→          title={
   651→            action === "approve"
   652→              ? "Approve expense"
   653→              : action === "reject"
   654→                ? "Reject expense"
   655→                : "Void expense"
   656→          }
   657→          description={
   658→            action === "approve" ? (
   659→              <>
   660→                The money is posted to the mess ledger (Dr Expenses / Cr Cash) and counts toward this month's formula.
   661→                <span className="mt-2 block font-medium">
   662→                  {detail.displayNumber} · {detail.totalFormatted}
   663→                </span>
   664→              </>
   665→            ) : action === "reject" ? (
   666→              <>
   667→                The record stays for the audit trail but no money moves. The submitter is notified with your reason.
   668→                <span className="mt-2 block font-medium">
   669→                  {detail.displayNumber} · {detail.totalFormatted}
   670→                </span>
   671→              </>
   672→            ) : (
   673→              <>
   674→                A reversal journal is posted — the money returns to cash and the expense stays visible as voided. Approved
   675→                expenses are never deleted.
   676→                <span className="mt-2 block font-medium">
   677→                  {detail.displayNumber} · {detail.totalFormatted}
   678→                </span>
   679→              </>
   680→            )
   681→          }
   682→          confirmLabel={action === "approve" ? "Approve" : action === "reject" ? "Reject" : "Void"}
   683→          tone={action === "approve" ? "primary" : "destructive"}
   684→          requireReason={action !== "approve"}
   685→          loading={acting}
   686→          onConfirm={(reason) => void runAction(action, reason)}
   687→        />
   688→      )}
   689→    </div>
   690→  );
   691→}
