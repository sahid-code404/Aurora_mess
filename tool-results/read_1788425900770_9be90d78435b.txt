   224→/** Full date-time format: "29 June 2026, 05:28 am" */
   225→function formatDate(iso: string) {
   226→  const d = new Date(iso);
   227→  const datePart = d.toLocaleDateString("en-GB", { day: "numeric", month: "long", year: "numeric" });
   228→  const timePart = d.toLocaleTimeString("en-US", { hour: "2-digit", minute: "2-digit", hour12: true });
   229→  return `${datePart}, ${timePart}`;
   230→}
   231→
   232→/** Format quantity + unit for display (e.g. "5 kg", "2 piece"). */
   233→function formatQuantity(qty: number, unit: string): string {
   234→  if (!qty && !unit) return "";
   235→  if (!unit) return String(qty);
   236→  if (!qty) return unit;
   237→  return `${qty} ${unit}`;
   238→}

   240→/**
   241→ * An expense is locked when its status is LOCKED, it's in the deletion queue,
   242→ * or it belongs to a past month (bills may have been generated against it).
   243→ */
   244→function isExpenseLocked(expense: Expense): boolean {
   245→  if (expense.status === "LOCKED") return true;
   246→  if (expense.deletedAt) return true; // in deletion queue — can't edit
   247→  const expDate = new Date(expense.expenseDate);
   248→  const now = new Date();
   249→  const expYM = expDate.getFullYear() * 12 + expDate.getMonth();
   250→  const todayYM = now.getFullYear() * 12 + now.getMonth();
   251→  return expYM < todayYM;
   252→}

   337→        form.append("proof", proof);

   339→      }

   341→    },
   342→    onSuccess: () => {
   343→      toast.success("Expense updated successfully");
   344→      closeForm();
   345→      qc.invalidateQueries({ queryKey: ["expenses"] });

  1117→  );

  1119→  const [customCategory, setCustomCategory] = useState(initialCustomCategory);
  1120→  const [unit, setUnit] = useState<string>(initialUnit);
  1121→  const [customUnit, setCustomUnit] = useState(initialCustomUnit);
