    95→/* ------------------------------- date helpers ------------------------------ */

    97→/** (year, monthIndex 0-11) → "YYYY-MM". */
    98→function monthKeyOf(year: number, month: number): string {
    99→  return `${year}-${String(month + 1).padStart(2, "0")}`;
   100→}
   101→
   102→function monthStartKey(year: number, month: number): string {
   103→  return `${monthKeyOf(year, month)}-01`;
   104→}
   105→
   106→function monthEndKey(year: number, month: number): string {
   107→  const lastDay = new Date(Date.UTC(year, month + 1, 0)).getUTCDate();
   108→  return `${monthKeyOf(year, month)}-${String(lastDay).padStart(2, "0")}`;
   109→}
   110→
   111→/** "September 2025". */
   112→function monthLongLabel(year: number, month: number): string {
   113→  return new Intl.DateTimeFormat("en-IN", { month: "long", year: "numeric", timeZone: "UTC" }).format(
   114→    new Date(Date.UTC(year, month, 1))
   115→  );
   116→}
   117→
   118→/** "Sep 2025" — compact KPI sub for non-current months. */
   119→function monthShortKey(year: number, month: number): string {
   120→  return new Intl.DateTimeFormat("en-IN", { month: "short", year: "numeric", timeZone: "UTC" }).format(
   121→    new Date(Date.UTC(year, month, 1))
   122→  );
   123→}
   124→
   125→function parseKey(key: string): Date {
   126→  const [y, m, d] = key.split("-").map(Number);
   127→  return new Date(Date.UTC(y, (m ?? 1) - 1, d ?? 1));
   128→}
   129→
   130→function weekdayShort(key: string): string {
   131→  return new Intl.DateTimeFormat("en-IN", { weekday: "short", timeZone: "UTC" }).format(parseKey(key));
   132→}
   133→
   134→function dayNum(key: string): string {
   135→  return String(parseKey(key).getUTCDate());
   136→}
   137→
   138→/** "Monday, 3 September". */
   139→function longDayLabel(key: string): string {
   140→  return new Intl.DateTimeFormat("en-IN", { weekday: "long", day: "numeric", month: "long", timeZone: "UTC" }).format(
   141→    parseKey(key)
   142→  );
   143→}
   144→
   145→/** Picker pill labels — relative for ±1 day, compact otherwise (BoardOps pattern). */
   146→function dayPillLabels(key: string, todayKey: string): { top: string; bottom: string } {
   147→  if (key === todayKey) return { top: "Today", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   148→  if (key === addDaysToKey(todayKey, -1)) return { top: "Yesterday", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   149→  if (key === addDaysToKey(todayKey, 1)) return { top: "Tomorrow", bottom: `${weekdayShort(key)}, ${formatDateInTz(key, "UTC")}` };
   150→  return { top: formatDateInTz(key, "UTC"), bottom: weekdayShort(key) };
   151→}

   157→function MiniChip({
   158→  tone = "neutral",
   159→  icon: Icon,
   160→  children,
   161→}: {
   162→  tone?: "success" | "warning" | "danger" | "primary" | "neutral";

   167→    success: "bg-success/15 text-success",
   168→    warning: "bg-warning/15 text-warning",
   169→    danger: "bg-danger/15 text-danger",
   170→    primary: "bg-primary/15 text-primary",
   171→    neutral: "bg-muted/80 text-muted-foreground",
   172→  };
   173→  return (
   174→    <span
   175→      className={cn(
   176→        "inline-flex items-center gap-0.5 whitespace-nowrap rounded-pill px-1.5 py-0.5 text-[10px] font-medium",
   177→        tones[tone]
   178→      )}
   179→    >
   180→      {Icon && <Icon className="size-2.5" aria-hidden />}
   181→      {children}
   182→    </span>
   183→  );
   184→}

   221→function AgendaDayRow({
   222→  dateKey,
   223→  todayKey,
   224→  entries,
   225→  tz,
   226→  now,
   227→  expanded,
   228→  guestsByInstance,
   229→  flash,
   230→  onToggleExpand,
   231→  onToggleMeal,
   232→}: {
   233→  dateKey: string;
   234→  todayKey: string;
   235→  entries: MealInstanceDto[];
   236→  tz: string;
   237→  now: number;
   238→  expanded: boolean;
   239→  guestsByInstance: Map<string, number>;
   240→  flash?: Record<string, Flash>;
   241→  onToggleExpand: () => void;
   242→  onToggleMeal: (meal: MealInstanceDto, next: "ON" | "OFF") => void;
   243→}) {
   244→  const isToday = dateKey === todayKey;
   245→  const onCount = entries.filter((m) => m.myState.effectiveState === "ON").length;
   246→  const offCount = entries.filter((m) => m.myState.effectiveState === "OFF").length;
   247→  const leaveCount = entries.filter((m) => m.myState.effectiveState === "ON_LEAVE").length;
   248→  const lockedCount = entries.filter((m) => m.myState.locked).length;
   249→  const adminCount = entries.filter((m) => m.myState.overridden).length;
   250→
   251→  return (
   252→    <GlassCard className="overflow-hidden">
   253→      <button
   254→        type="button"
   255→        onClick={onToggleExpand}
   256→        aria-expanded={expanded}
   257→        className="flex w-full cursor-pointer items-center gap-3 p-3 text-left transition-colors hover:bg-foreground/[0.04] dark:hover:bg-white/[0.04]"
   258→      >
   259→        <div
   260→          className={cn(
   261→            "flex size-11 shrink-0 flex-col items-center justify-center rounded-xl",
   262→            isToday ? "bg-primary/15 text-primary" : "glass-inset text-muted-foreground"
   263→          )}
   264→        >
   265→          <span className="text-[10px] font-bold leading-none">{weekdayShort(dateKey).toUpperCase()}</span>
   266→          <span className={cn("mt-0.5 text-sm font-bold leading-none", isToday ? "text-primary" : "text-foreground")}>
   267→            {dayNum(dateKey)}
   268→          </span>
   269→        </div>
   270→        <div className="min-w-0 flex-1">
   271→          <p className={cn("truncate text-sm font-medium", isToday && "text-primary")}>
   272→            {isToday ? "Today" : longDayLabel(dateKey)}
   273→          </p>
   274→          <div className="mt-1 flex flex-wrap items-center gap-1.5">
   275→            {onCount > 0 && (
   276→              <MiniChip tone="success" icon={Check}>
   277→                {onCount} ON
   278→              </MiniChip>
   279→            )}
   280→            {offCount > 0 && (
   281→              <MiniChip tone="warning" icon={X}>
   282→                {offCount} OFF
   283→              </MiniChip>
   284→            )}
   285→            {leaveCount > 0 && <MiniChip tone="neutral" icon={Plane}>{leaveCount} on leave</MiniChip>}
   286→            {lockedCount > 0 && (
   287→              <MiniChip tone="danger" icon={Lock}>
   288→                {lockedCount}
   289→              </MiniChip>
   290→            )}
   291→            {adminCount > 0 && (
   292→              <MiniChip tone="primary" icon={ShieldCheck}>
   293→                {adminCount} admin
   294→              </MiniChip>
   295→            )}
   296→          </div>
   297→        </div>
   298→        <motion.span
   299→          animate={{ rotate: expanded ? 180 : 0 }}
   300→          transition={{ duration: 0.2 }}
   301→          className="flex shrink-0 items-center text-muted-foreground"
   302→        >
   303→          <ChevronDown className="size-4" aria-hidden />
   304→        </motion.span>
   305→      </button>
   306→
   307→      <AnimatePresence initial={false}>
   308→        {expanded && (
   309→          <motion.div
   310→            initial={{ height: 0, opacity: 0 }}
   311→            animate={{ height: "auto", opacity: 1 }}
   312→            exit={{ height: 0, opacity: 0 }}
   313→            transition={{ duration: 0.25, ease: [0.22, 1, 0.36, 1] }}
   314→            className="overflow-hidden"
   315→          >
   316→            <div className="space-y-2 px-3 pb-3">
   317→              {entries.map((meal) => (
   318→                <AgendaMealRow
   319→                  key={meal.id}
   320→                  meal={meal}
   321→                  tz={tz}
   322→                  now={now}
   323→                  guests={guestsByInstance.get(meal.id) ?? 0}
   324→                  flash={flash?.[meal.id]}
   325→                  onToggle={onToggleMeal}
   326→                />
   327→              ))}
   328→            </div>
   329→          </motion.div>
   330→        )}
   331→      </AnimatePresence>
   332→    </GlassCard>
   333→  );
   334→}

   336→/* ---------------------------- agenda meal row ------------------------------ */
   337→
   338→function AgendaMealRow({
   339→  meal,
   340→  tz,
   341→  now,
   342→  guests,
   343→  flash,
   344→  onToggle,
   345→}: {
   346→  meal: MealInstanceDto;
   347→  tz: string;
   348→  now: number;
   349→  guests: number;
   350→  flash?: Flash;
   351→  onToggle: (meal: MealInstanceDto, next: "ON" | "OFF") => void;
   352→}) {
   353→  const state = meal.myState.effectiveState;
   354→  const overridden = meal.myState.overridden;
   355→  const locked = meal.myState.locked || new Date(meal.cutoffAt).getTime() <= now;
   356→  const canToggle = !locked && !overridden && (state === "ON" || state === "OFF");
   357→  const cutoffLeft = new Date(meal.cutoffAt).getTime() - now;
   358→
   359→  return (
   360→    <div className="glass-inset rounded-lg">
   361→      <div className="flex items-center gap-3 p-2.5">
   362→        <span
   363→          aria-hidden
   364→          className={cn("flex size-10 shrink-0 items-center justify-center rounded-lg border", mealTint(meal.colorToken))}
   365→        >
   366→          <MealIcon name={meal.icon} className="size-[18px]" />
   367→        </span>
   368→        <div className="min-w-0 flex-1">
   369→          <p className="truncate text-sm font-medium">{meal.name}</p>
   370→          <div className="mt-0.5 flex flex-wrap items-center gap-x-2 gap-y-1 text-[11px] text-muted-foreground">
   371→            <span className="kpi-num">{formatWindowInTz(meal.serviceWindow.startAt, meal.serviceWindow.endAt, tz)}</span>
   372→            {canToggle && cutoffLeft > 0 && (
   373→              <>
   374→                <span aria-hidden>·</span>
   375→                <span className="kpi-num">{countdownLabel(cutoffLeft)}</span>
   376→              </>
   377→            )}
   378→            {locked && <Lock className="size-3 text-danger" aria-hidden />}
   379→            {overridden && (
   380→              <MiniChip tone="primary" icon={ShieldCheck}>
   381→                Admin
   382→              </MiniChip>
   383→            )}
   384→            {state === "ON_LEAVE" && <MiniChip tone="neutral" icon={Plane}>On leave</MiniChip>}
   385→            {state === "NOT_AVAILABLE" && meal.myState.effectiveReason in NOT_AVAILABLE_REASONS && (
   386→              <span className="font-medium">{NOT_AVAILABLE_REASONS[meal.myState.effectiveReason]}</span>
   387→            )}
   388→            {guests > 0 && (
   389→              <span className="kpi-num inline-flex items-center gap-1 font-medium text-primary">
   390→                <Users className="size-3" aria-hidden />
   391→                {guests}
   392→              </span>
   393→            )}
   394→          </div>
   395→        </div>
   396→        <div className="flex shrink-0 items-center gap-1.5">
   397→          {canToggle ? (
   398→            <GlassToggle
   399→              checked={state === "ON"}
   400→              onChange={(next) => onToggle(meal, next ? "ON" : "OFF")}
   401→              label={`${meal.name} on ${meal.serviceDate}: ${stateLabel(state)}`}
   402→            />
   403→          ) : (
   404→            <>
   405→              {overridden && <StatusBadge status="ADMIN_OVERRIDE" />}
   406→              <StatusBadge status={state} label={stateLabel(state)} />
   407→              {locked && <StatusBadge status="LOCKED" />}
   408→            </>
   409→          )}
   410→        </div>
   411→      </div>
   412→
   413→      <AnimatePresence>
   414→        {flash && (
   415→          <motion.div
   416→            initial={{ opacity: 0, height: 0 }}
   417→            animate={{ opacity: 1, height: "auto" }}
   418→            exit={{ opacity: 0, height: 0 }}
   419→            transition={{ duration: 0.22 }}
   420→            className="overflow-hidden"
   421→          >
   422→            <div className="px-2.5 pb-2.5">
   423→              <FormNotice tone={flash.tone === "info" ? "info" : flash.tone}>{flash.text}</FormNotice>
   424→            </div>
   425→          </motion.div>
   426→        )}
   427→      </AnimatePresence>
   428→    </div>
   429→  );
   430→}

   605→  async function handleToggle(meal: MealInstanceDto, next: "ON" | "OFF") {
   606→    const queries = queryClient.getQueriesData<MealsCacheValue>({
   607→      queryKey: ["api", "/api/v1/meals"],

   611→    let current: MealInstanceDto | undefined;
   612→    for (const { value } of snapshots) {
   613→      current = value?.data.find((m) => m.id === meal.id);
   614→      if (current) break;
   615→    }
   616→    if (!current) return;
   617→
   618→    const expectedVersion = current.myState.version;
   619→
   620→    // Optimistically flip every cached copy of this row.
   621→    for (const { key } of snapshots) {
   622→      queryClient.setQueryData<MealsCacheValue>(key, (old) =>
   623→        patchCacheRows(old, meal.id, (m) => ({
   624→          ...m,
   625→          myState: { ...m.myState, effectiveState: next } as MealMyState,
   626→        }))
   627→      );
   628→    }
   629→
   630→    try {
   631→      const res = await apiJson<ToggleResponse>(`/api/v1/meals/${meal.id}/toggle`, "POST", {
   632→        state: next,
   633→        expectedVersion,
   634→      });
   635→      // Authoritative response wins.
   636→      for (const { key } of snapshots) {
   637→        queryClient.setQueryData<MealsCacheValue>(key, (old) =>
   638→          patchCacheRows(old, meal.id, (m) => ({
   639→            ...m,
   640→            myState: {
   641→              ...m.myState,
   642→              effectiveState: res.state,
   643→              effectiveReason: res.effectiveReason,
   644→              locked: res.locked,
   645→              version: res.version,
   646→            } as MealMyState,
   647→          }))
   648→        );
   649→      }
   650→      invalidate([RESIDENT_KEYS.dashboard]);
   651→    } catch (err) {
   652→      // Rollback — the toggle springs back (spec §114).
   653→      for (const { key, value } of snapshots) {
   654→        queryClient.setQueryData<MealsCacheValue>(key, value);
   655→      }

   657→        if (err.code === "MEAL_CUTOFF_PASSED") {
   658→          flashFor(meal.id, {
   659→            tone: "warning",
   660→            text: `${meal.name} is already locked. Admin can still override it.`,
   661→          });
   662→        } else if (err.code === "RESOURCE_CHANGED") {
   663→          flashFor(meal.id, { tone: "info", text: "This meal was just changed. Refreshing…" });

   666→          flashFor(meal.id, { tone: "warning", text: "This meal isn't available to change right now." });
   667→        } else {
   668→          flashFor(meal.id, { tone: "danger", text: err.message });
   669→        }
   670→      } else {
   671→        flashFor(meal.id, { tone: "danger", text: friendlyError(err) });
